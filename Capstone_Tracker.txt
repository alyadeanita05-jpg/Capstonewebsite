function generateInviteCode() {
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  let result = 'CAP-';
  for (let i = 0; i < 4; i++) {
    result += chars.charAt(Math.floor(Math.random() * chars.length));
  }
  return result;
}

function formatDateID(dateStr) {
  if (!dateStr) return '-';
  try {
    const d = new Date(dateStr);
    if (isNaN(d.getTime())) return dateStr;
    return d.toLocaleDateString('id-ID', {
      day: 'numeric',
      month: 'short',
      year: 'numeric'
    });
  } catch {
    return dateStr;
  }
}

function formatDateTimeID(timestamp) {
  if (!timestamp) return 'Baru saja';
  try {
    const d = timestamp?.toDate ? timestamp.toDate() : new Date(timestamp);
    if (isNaN(d.getTime())) return 'Baru saja';
    return d.toLocaleString('id-ID', {
      day: 'numeric',
      month: 'short',
      hour: '2-digit',
      minute: '2-digit'
    });
  } catch {
    return 'Baru saja';
  }
}

function getInitials(name) {
  if (!name) return 'U';
  const parts = name.trim().split(' ');
  if (parts.length === 1) return parts[0].substring(0, 2).toUpperCase();
  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
}

function getStatusBadgeStyle(status) {
  switch (status) {
    case 'Selesai':
    case 'Selesai Direvisi':
      return 'bg-emerald-50 text-emerald-700 border-emerald-200';
    case 'Sedang Dikerjakan':
    case 'Sedang Direvisi':
      return 'bg-amber-50 text-amber-700 border-amber-200';
    case 'Menunggu Review':
      return 'bg-pink-50 text-pink-700 border-pink-200';
    case 'Perlu Revisi':
      return 'bg-red-50 text-red-700 border-red-200';
    default:
      return 'bg-slate-100 text-slate-600 border-slate-200';
  }
}

function getPriorityBadgeStyle(priority) {

  }
}

// Compute individual member progress in a task
function computeMemberProgress(memberWork) {
  if (!memberWork) return 0;
  const checklists = memberWork.checklists || [];
  if (checklists.length > 0) {
    const done = checklists.filter((c) => c.done).length;
    return Math.round((done / checklists.length) * 100);
  }
  return typeof memberWork.manualProgress === 'number' ? memberWork.manualProgress : 0;
}

// Compute overall task progress from assigned members or task status
function computeTaskProgress(task) {
  if (task.status === 'Selesai') return 100;
  const assignedIds = task.assignedMemberIds || [];
  const memberWorkMap = task.memberWork || {};
  if (assignedIds.length === 0) {
    if (task.status === 'Menunggu Review') return 90;
    if (task.status === 'Sedang Dikerjakan') return 40;
    if (task.status === 'Perlu Revisi') return 65;
    return 0;
  }
  const totalPct = assignedIds.reduce((acc, uid) => {
    return acc + computeMemberProgress(memberWorkMap[uid]);
  }, 0);
  return Math.round(totalPct / assignedIds.length);
}

function AvatarBadge({ name, color = 'bg-pink-500', url = '', size = 'md' }) {
  const sizeClasses = {
    xs: 'w-6 h-6 text-[10px]',
    sm: 'w-8 h-8 text-xs',
    md: 'w-9 h-9 text-xs',
    lg: 'w-12 h-12 text-base',
    xl: 'w-16 h-16 text-xl'
  }[size] || 'w-9 h-9 text-xs';

  if (url) {
    return (
      <img
        src={url}
        alt={name || 'Avatar'}
        className={`${sizeClasses} rounded-full object-cover border border-white shadow-xs shrink-0`}
      />
    );
  }

  return (
    <div
      className={`${sizeClasses} ${color} rounded-full flex items-center justify-center text-white font-bold shadow-xs shrink-0 select-none`}
    >
      {getInitials(name)}
    </div>
  );
}

function ToastBanner({ toast, onClose }) {
  if (!toast) return null;
  const isError = toast.type === 'error';
  return (
    <div className="fixed bottom-5 right-5 z-[100] max-w-sm animate-bounce-in">
      <div
        className={`flex items-center gap-3 px-4 py-3 rounded-2xl shadow-xl border ${
          isError
            ? 'bg-red-600 text-white border-red-500'
            : 'bg-slate-900 text-white border-slate-800'
        }`}
      >
        {isError ? (
          <AlertCircle className="w-5 h-5 text-red-200 shrink-0" />
        ) : (
          <CheckCircle2 className="w-5 h-5 text-pink-400 shrink-0" />
        )}
        <p className="text-sm font-medium flex-1">{toast.message}</p>
        <button onClick={onClose} className="text-white/70 hover:text-white">
          <X className="w-4 h-4" />
        </button>
      </div>
    </div>
  );
}

function ProgressBar({ value = 0, colorClass = 'bg-pink-500', heightClass = 'h-2' }) {
  const safeVal = Math.max(0, Math.min(100, Number(value) || 0));
  return (
    <div className={`w-full bg-slate-100 rounded-full overflow-hidden ${heightClass}`}>
      <div
        className={`${colorClass} ${heightClass} rounded-full transition-all duration-500 ease-out`}
        style={{ width: `${safeVal}%` }}
      />
    </div>
  );
}

function AuthScreen({ showToast, initialInviteCode }) {
  const [mode, setMode] = useState('login'); // 'login' | 'register' | 'forgot'
  const [name, setName] = useState('');
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [loading, setLoading] = useState(false);

  const handleEmailAuth = async (e) => {
    e.preventDefault();
    setLoading(true);
    try {
      if (mode === 'login') {
        await signInWithEmailAndPassword(auth, email.trim(), password);
        showToast('Berhasil masuk ke Capstone Tracker!');
      } else if (mode === 'register') {
        if (!name.trim()) {
          showToast('Nama lengkap wajib diisi.', 'error');
          setLoading(false);
          return;
        }
        const cred = await createUserWithEmailAndPassword(auth, email.trim(), password);
        await updateProfile(cred.user, { displayName: name.trim() });
        await setDoc(doc(db, 'artifacts', appId, 'users', cred.user.uid, 'profile', 'main'), {
          uid: cred.user.uid,
          name: name.trim(),
          email: email.trim(),
          nim: '',
          bio: 'Anggota Tim Capstone Project',
          avatarColor: AVATAR_COLORS[Math.floor(Math.random() * AVATAR_COLORS.length)],
          avatarUrl: '',
          activeGroupId: null,
          groupIds: [],
          createdAt: new Date().toISOString()
        });
        showToast('Akun berhasil dibuat! Selamat datang.');
      } else if (mode === 'forgot') {
        await sendPasswordResetEmail(auth, email.trim());
        showToast('Link reset password telah dikirim ke email Anda.');
        setMode('login');
      }
    } catch (err) {
      let msg = 'Terjadi kesalahan autentikasi.';
      if (err.code === 'auth/invalid-credential' || err.code === 'auth/wrong-password') {
        msg = 'Email atau password salah.';
      } else if (err.code === 'auth/email-already-in-use') {
        msg = 'Email sudah terdaftar. Silakan login.';
      } else if (err.code === 'auth/weak-password') {
        msg = 'Password minimal 6 karakter.';
      } else if (err.code === 'auth/user-not-found') {
        msg = 'Akun dengan email ini tidak ditemukan.';
      }
      showToast(msg, 'error');
    } finally {
      setLoading(false);
    }
  };

  const handleInstantGuestDemo = async () => {
    setLoading(true);
    try {
      const cred = await signInAnonymously(auth);
      const guestNum = Math.floor(100 + Math.random() * 900);
      const guestName = `Mahasiswa #${guestNum}`;
      await setDoc(doc(db, 'artifacts', appId, 'users', cred.user.uid, 'profile', 'main'), {
        uid: cred.user.uid,
        name: guestName,
        email: `mahasiswa${guestNum}@capstone.ac.id`,
        nim: '',
        bio: 'Anggota Tim Capstone Project',
        avatarColor: 'bg-pink-500',
        avatarUrl: '',
        activeGroupId: null,
        groupIds: [],
        createdAt: new Date().toISOString()
      });
      showToast(`Masuk sebagai ${guestName}`);
    } catch (err) {
      showToast('Gagal memulai sesi cepat.', 'error');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="min-h-screen bg-gradient-to-br from-pink-50 via-white to-rose-50 flex flex-col justify-center py-12 px-4 sm:px-6 lg:px-8">
      <div className="max-w-md w-full mx-auto">
        {/* Header Brand */}
        <div className="text-center mb-8">
          <div className="w-16 h-16 rounded-3xl bg-gradient-to-br from-pink-500 to-rose-600 flex items-center justify-center text-white shadow-xl shadow-pink-200 mx-auto mb-4">
            <FolderKanban className="w-8 h-8" />
          </div>
          <h1 className="text-3xl font-extrabold text-slate-900 tracking-tight">
            CAPSTONE TRACKER
          </h1>
          <p className="text-sm text-pink-600 font-medium mt-1.5">
            Organize Your Tasks, Track Your Progress, Achieve Together.
          </p>
          {initialInviteCode && (
            <div className="mt-3 inline-flex items-center gap-2 px-3.5 py-1.5 rounded-full bg-pink-100 text-pink-800 text-xs font-semibold">
              <Sparkles className="w-3.5 h-3.5" />
              Undangan Kelompok: {initialInviteCode}
            </div>
          )}
        </div>

        {/* Auth Card */}
        <div className="bg-white rounded-3xl shadow-xl shadow-pink-100/60 border border-pink-100 p-7 sm:p-8">
          {mode !== 'forgot' && (
            <div className="grid grid-cols-2 gap-1.5 p-1 bg-slate-100 rounded-2xl mb-6">
              <button
                type="button"
                onClick={() => setMode('login')}
                className={`py-2 text-sm font-semibold rounded-xl transition ${
                  mode === 'login'
                    ? 'bg-white text-pink-600 shadow-xs'
                    : 'text-slate-600 hover:text-slate-900'
                }`}
              >
                Masuk
              </button>
              <button
                type="button"
                onClick={() => setMode('register')}
                className={`py-2 text-sm font-semibold rounded-xl transition ${
                  mode === 'register'
                    ? 'bg-white text-pink-600 shadow-xs'
                    : 'text-slate-600 hover:text-slate-900'
                }`}
              >
                Daftar Akun
              </button>
            </div>
          )}

          {mode === 'forgot' && (
            <div className="mb-6">
              <button
                type="button"
                onClick={() => setMode('login')}
                className="text-xs font-semibold text-pink-600 hover:text-pink-700 flex items-center gap-1 mb-2"
              >
                <ChevronLeft className="w-4 h-4" /> Kembali ke Login
              </button>
              <h2 className="text-xl font-bold text-slate-900">Lupa Password</h2>
              <p className="text-xs text-slate-500 mt-1">
                Masukkan email terdaftar Anda untuk menerima tautan pemulihan password.
              </p>
            </div>
          )}

          <form onSubmit={handleEmailAuth} className="space-y-4">
            {mode === 'register' && (
              <div>
                <label className="block text-xs font-semibold text-slate-700 mb-1.5">
                  Nama Lengkap
                </label>
                <input
                  type="text"
                  required
                  value={name}
                  onChange={(e) => setName(e.target.value)}
                  placeholder="Contoh: Alya Putri"
                  className="w-full px-4 py-2.5 rounded-xl border border-slate-200 text-sm focus:outline-none focus:ring-2 focus:ring-pink-400 focus:border-transparent"
                />
              </div>
            )}

            <div>
              <label className="block text-xs font-semibold text-slate-700 mb-1.5">

            </div>

            {mode !== 'forgot' && (
              <div>
                <div className="flex items-center justify-between mb-1.5">
                  <label className="block text-xs font-semibold text-slate-700">Password</label>
                  {mode === 'login' && (
                    <button
                      type="button"

                      className="text-xs font-semibold text-pink-600 hover:text-pink-700"
                    >
                      Lupa password?
                    </button>
                  )}
                </div>
                <input
                  type="password"
                  required
                  minLength={6}
                  value={password}
                  onChange={(e) => setPassword(e.target.value)}
                  placeholder="Minimal 6 karakter"
                  className="w-full px-4 py-2.5 rounded-xl border border-slate-200 text-sm focus:outline-none focus:ring-2 focus:ring-pink-400 focus:border-transparent"
                />
              </div>
            )}

            <button
              type="submit"
              disabled={loading}
              className="w-full py-3 rounded-xl bg-gradient-to-r from-pink-500 to-rose-600 hover:from-pink-600 hover:to-rose-700 text-white font-semibold text-sm shadow-md shadow-pink-200 transition disabled:opacity-60"
            >
              {loading
                ? 'Memproses...'
                : mode === 'login'
                ? 'Masuk ke Ruang Kerja'
                : mode === 'register'
                ? 'Daftar Sekarang'
                : 'Kirim Link Reset Password'}
            </button>
          </form>

      const foundDoc = groupsSnap.docs.find(
        (d) => (d.data().inviteCode || '').toUpperCase() === cleanCode
      );

      if (!foundDoc) {
        showToast('Kode undangan tidak ditemukan. Periksa kembali kode dari ketua kelompok.', 'error');
        setLoading(false);
        return;
      }

      const gid = foundDoc.id;
      const gData = foundDoc.data();

      // Add user to group members
      await setDoc(
        doc(db, 'artifacts', appId, 'public', 'data', `grp_${gid}_members`, authUser.uid),
        {
          uid: authUser.uid,
          name: userProfile?.name || 'Anggota Kelompok',
          email: userProfile?.email || authUser.email || '',
          role: gData.leaderId === authUser.uid ? 'Ketua Kelompok' : 'Anggota Kelompok',
          avatarColor: userProfile?.avatarColor || 'bg-rose-500',
          avatarUrl: userProfile?.avatarUrl || '',
          joinedAt: new Date().toISOString()
        },
        { merge: true }
      );

      await updateDoc(doc(db, 'artifacts', appId, 'users', authUser.uid, 'profile', 'main'), {
        activeGroupId: gid,
        groupIds: arrayUnion(gid)
      });

      await addDoc(
        collection(db, 'artifacts', appId, 'public', 'data', `grp_${gid}_activities`),
        {
          userId: authUser.uid,
          userName: userProfile?.name || 'Anggota Baru',
          avatarColor: userProfile?.avatarColor || 'bg-rose-500',
          action: `bergabung ke dalam kelompok`,
          category: 'group',
          createdAt: new Date().toISOString()
        }
      );

      showToast(`Berhasil bergabung ke kelompok "${gData.name}"!`);
      onGroupJoined(gid);
    } catch (err) {
      console.error(err);
      showToast('Gagal bergabung ke kelompok.', 'error');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="min-h-screen bg-slate-50 flex flex-col justify-center py-12 px-4 sm:px-6">
      <div className="max-w-lg w-full mx-auto">
        <div className="flex items-center justify-between mb-6">
          <div className="flex items-center gap-2.5">
            <div className="w-10 h-10 rounded-2xl bg-pink-500 flex items-center justify-center text-white shadow-md shadow-pink-200">
              <FolderKanban className="w-5 h-5" />
            </div>
            <div>
              <h1 className="font-bold text-slate-900 text-base">Halo, {userProfile?.name}!</h1>
              <p className="text-xs text-slate-500">Pilih atau buat kelompok Capstone Anda</p>
            </div>
          </div>
          <button
            onClick={() => signOut(auth)}
            className="text-xs font-semibold text-slate-500 hover:text-red-600 flex items-center gap-1 px-3 py-1.5 rounded-xl hover:bg-red-50 transition"

            <LogOut className="w-3.5 h-3.5" /> Keluar

        <div className="bg-white rounded-3xl shadow-xl shadow-slate-200/60 border border-slate-100 p-6 sm:p-8">
          <div className="grid grid-cols-2 gap-2 p-1.5 bg-slate-100 rounded-2xl mb-6">
            <button

                type="submit"

              >

          ) : (

                />
                <p className="text-xs text-slate-500 mt-2">
                  Minta kode undangan atau link undangan dari Ketua Kelompok Anda.
                </p>
              </div>

              <button
                type="submit"
                disabled={loading}
                className="w-full py-3 rounded-xl bg-gradient-to-r from-pink-500 to-rose-600 hover:from-pink-600 hover:to-rose-700 text-white font-semibold text-sm shadow-md shadow-pink-200 transition disabled:opacity-60"
              >
                {loading ? 'Memeriksa Kode...' : 'Gabung ke Kelompok'}

            </form>
          )}
        </div>
      </div>
    </div>
  );
}

                      {formatDateTimeID(act.createdAt)}
                    </span>
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>
      </div>

      {/* Add Course Modal */}
      {showCourseModal && (
        <CourseFormModal
          activeGroupId={activeGroupId}
          onClose={() => setShowCourseModal(false)}
          logActivity={logActivity}
          triggerNotification={triggerNotification}
          showToast={showToast}

}

}) {
  const [modalCourse, setModalCourse] = useState(null); // null | 'new' | courseObj

  const handleDeleteCourse = async (course) => {
    try {

      setConfirmDeleteCourse(null);
    } catch (err) {
      showToast('Gagal menghapus mata kuliah.', 'error');
    }
  };

  return (
    <div className="space-y-6">
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
        <div>
          <h1 className="text-2xl font-extrabold text-slate-900">Mata Kuliah Capstone</h1>
          <p className="text-sm text-slate-500">
            Kelola seluruh mata kuliah, dosen pengampu, dan struktur minggu pengerjaan.
          </p>
        </div>
        <button
          onClick={() => setModalCourse('new')}
          className="px-4 py-2.5 rounded-2xl bg-gradient-to-r from-pink-500 to-rose-600 text-white text-sm font-semibold shadow-md shadow-pink-200 flex items-center gap-2 self-start"
        >
          <Plus className="w-4 h-4" /> Tambah Mata Kuliah
        </button>
      </div>

      {courses.length === 0 ? (
        <div className="bg-white rounded-3xl border border-dashed border-pink-200 p-12 text-center">
          <BookOpen className="w-10 h-10 text-pink-400 mx-auto mb-3" />
          <h3 className="font-bold text-slate-800">Daftar Mata Kuliah Masih Kosong</h3>
          <p className="text-xs text-slate-500 max-w-md mx-auto mt-1 mb-4">
            Tambahkan mata kuliah Capstone yang sedang diambil kelompok Anda.
          </p>
          <button
            onClick={() => setModalCourse('new')}
            className="px-5 py-2.5 rounded-xl bg-pink-500 text-white text-xs font-semibold"
          >
            + Tambah Mata Kuliah
          </button>
        </div>
      ) : (
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-5">
          {courses.map((course) => {
            const cWeeks = weeks.filter((w) => w.courseId === course.id);
            const cTasks = tasks.filter((t) => t.courseId === course.id);
            const cDone = cTasks.filter((t) => t.status === 'Selesai').length;
            const cProgress =
              cTasks.length === 0
                ? 0
                : Math.round(
                    cTasks.reduce((sum, t) => sum + computeTaskProgress(t), 0) / cTasks.length
                  );

            return (
              <div
                key={course.id}
                className="bg-white rounded-3xl border border-slate-100 shadow-xs hover:shadow-md transition overflow-hidden flex flex-col justify-between"
              >
                <div>
                  <div
                    className="h-2 w-full"
                    style={{ backgroundColor: course.color || '#EC4899' }}
                  />
                  <div className="p-5">
                    <div className="flex items-start justify-between gap-2">
                      <span className="text-[11px] font-bold px-2.5 py-0.5 rounded-full bg-pink-50 text-pink-700">
                        {course.semester} • {course.academicYear}
                      </span>
                      <div className="flex items-center gap-1">
                        <button
                          onClick={() => setModalCourse(course)}
                          className="p-1.5 rounded-lg text-slate-400 hover:text-pink-600 hover:bg-pink-50"
                          title="Edit Mata Kuliah"
                        >
                          <Edit3 className="w-4 h-4" />
                        </button>
                        <button
                          onClick={() => setConfirmDeleteCourse(course)}
                          className="p-1.5 rounded-lg text-slate-400 hover:text-red-600 hover:bg-red-50"
                          title="Hapus Mata Kuliah"
                        >
                          <Trash2 className="w-4 h-4" />
                        </button>
                      </div>
                    </div>

                    <h3 className="font-bold text-slate-900 text-lg mt-2">{course.name}</h3>
                    <p className="text-xs font-medium text-slate-500 mt-0.5">
                      Dosen: {course.lecturer || '-'}
                    </p>
                    <p className="text-xs text-slate-600 mt-2.5 line-clamp-2">
                      {course.description || 'Tidak ada deskripsi tambahan.'}
                    </p>

                    <div className="mt-4">
                      <div className="flex justify-between text-xs mb-1">
                        <span className="text-slate-500">Progres</span>
                        <span className="font-bold text-pink-600">{cProgress}%</span>
                      </div>
                      <ProgressBar value={cProgress} />
                    </div>

                    <div className="flex items-center justify-between mt-4 pt-3 border-t border-slate-100 text-xs text-slate-500">
                      <span>{cWeeks.length} Week</span>
                      <span>
                        {cDone}/{cTasks.length} Tugas Selesai
                      </span>
                    </div>
                  </div>
                </div>

                <div className="px-5 pb-5">
                  <button
                    onClick={() => onOpenCourse(course.id)}
                    className="w-full py-2.5 rounded-xl bg-slate-900 hover:bg-pink-600 text-white text-xs font-semibold transition flex items-center justify-center gap-1.5"
                  >
                    <span>Buka Detail & Week</span>
                    <ArrowRight className="w-3.5 h-3.5" />
                  </button>
                </div>
              </div>
            );
          })}
        </div>
      )}

      {modalCourse && (
        <CourseFormModal
          existingCourse={modalCourse === 'new' ? null : modalCourse}
          activeGroupId={activeGroupId}
          onClose={() => setModalCourse(null)}
          logActivity={logActivity}
          triggerNotification={triggerNotification}
          showToast={showToast}
        />
      )}

      {confirmDeleteCourse && (
        <ConfirmDialog
          title="Hapus Mata Kuliah?"
          message={`Anda yakin ingin menghapus "${confirmDeleteCourse.name}"?`}
          confirmText="Ya, Hapus"
          onConfirm={() => handleDeleteCourse(confirmDeleteCourse)}
          onCancel={() => setConfirmDeleteCourse(null)}
        />
      )}
    </div>
  );
}

function CourseFormModal({
  existingCourse = null,
  activeGroupId,
  onClose,
  logActivity,
  triggerNotification,
  showToast
}) {
  const [name, setName] = useState(existingCourse?.name || '');
  const [lecturer, setLecturer] = useState(existingCourse?.lecturer || '');
  const [description, setDescription] = useState(existingCourse?.description || '');
  const [semester, setSemester] = useState(existingCourse?.semester || 'Ganjil');
  const [academicYear, setAcademicYear] = useState(existingCourse?.academicYear || '2026/2027');
  const [color, setColor] = useState(existingCourse?.color || '#EC4899');
  const [saving, setSaving] = useState(false);

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (!name.trim()) return;
    setSaving(true);
    try {
      const payload = {
        name: name.trim(),
        lecturer: lecturer.trim(),
        description: description.trim(),
        semester: semester.trim(),
        academicYear: academicYear.trim(),
        color
      };

      if (existingCourse) {
        await updateDoc(
          doc(
            db,
            'artifacts',
            appId,
            'public',
            'data',
            `grp_${activeGroupId}_courses`,
            existingCourse.id
          ),
          payload
        );
        await logActivity(`memperbarui mata kuliah "${payload.name}"`, 'course', existingCourse.id);
        showToast('Mata kuliah berhasil diperbarui.');
      } else {
        const ref = await addDoc(
          collection(db, 'artifacts', appId, 'public', 'data', `grp_${activeGroupId}_courses`),
          {
            ...payload,
            createdAt: new Date().toISOString()
          }
        );
        // Automatically create Week 1 so user can start adding tasks right away
        await addDoc(
          collection(db, 'artifacts', appId, 'public', 'data', `grp_${activeGroupId}_weeks`),
          {
            courseId: ref.id,
            title: 'Week 1: Inisiasi & Pemahaman Materi',
            description: 'Minggu pertama pengerjaan tugas mata kuliah.',
            order: 1,
            createdAt: new Date().toISOString()
          }
        );
        await logActivity(`menambahkan mata kuliah baru "${payload.name}"`, 'course', ref.id);
        await triggerNotification({
          title: 'Mata Kuliah Baru',
          message: `Mata kuliah "${payload.name}" telah ditambahkan ke kelompok.`,
          type: 'course'
        });
        showToast('Mata kuliah berhasil ditambahkan!');
      }
      onClose();
    } catch (err) {
      showToast('Gagal menyimpan mata kuliah.', 'error');
    } finally {
      setSaving(false);
    }
  };

  return (
    <div className="fixed inset-0 z-50 bg-slate-900/50 backdrop-blur-xs flex items-center justify-center p-4">
      <div className="bg-white rounded-3xl max-w-lg w-full p-6 shadow-2xl border border-slate-100 max-h-[90vh] overflow-y-auto">
        <div className="flex items-center justify-between mb-5">
          <h2 className="text-lg font-bold text-slate-900">
            {existingCourse ? 'Edit Mata Kuliah' : 'Tambah Mata Kuliah Baru'}
          </h2>
          <button onClick={onClose} className="p-1.5 rounded-xl text-slate-400 hover:bg-slate-100">
            <X className="w-5 h-5" />
          </button>
        </div>

        <form onSubmit={handleSubmit} className="space-y-4">
          <div>
            <label className="block text-xs font-semibold text-slate-700 mb-1">
              Nama Mata Kuliah *
            </label>
            <input
              type="text"
              required
              value={name}
              onChange={(e) => setName(e.target.value)}
              placeholder="Contoh: Business Valuation / Project Management"
              className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm focus:ring-2 focus:ring-pink-400 focus:outline-none"
            />
          </div>

          <div>
            <label className="block text-xs font-semibold text-slate-700 mb-1">
              Nama Dosen Pengampu
            </label>
            <input
              type="text"
              value={lecturer}
              onChange={(e) => setLecturer(e.target.value)}
              placeholder="Contoh: Dr. Rina Kartika, S.E., M.Sc."
              className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm focus:ring-2 focus:ring-pink-400 focus:outline-none"
            />
          </div>

          <div className="grid grid-cols-2 gap-3">
            <div>
              <label className="block text-xs font-semibold text-slate-700 mb-1">Semester</label>
              <select
                value={semester}
                onChange={(e) => setSemester(e.target.value)}
                className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm focus:ring-2 focus:ring-pink-400 focus:outline-none"
              >
                <option value="Ganjil">Ganjil</option>
                <option value="Genap">Genap</option>
                <option value="Antara">Semester Antara</option>
              </select>
            </div>
            <div>
              <label className="block text-xs font-semibold text-slate-700 mb-1">
                Tahun Akademik
              </label>
              <input
                type="text"
                value={academicYear}
                onChange={(e) => setAcademicYear(e.target.value)}
                placeholder="2026/2027"
                className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm focus:ring-2 focus:ring-pink-400 focus:outline-none"
              />
            </div>
          </div>

          <div>
            <label className="block text-xs font-semibold text-slate-700 mb-1">
              Deskripsi Mata Kuliah
            </label>
            <textarea
              rows={3}
              value={description}
              onChange={(e) => setDescription(e.target.value)}
              placeholder="Tujuan pembelajaran atau ruang lingkup proyek Capstone pada mata kuliah ini..."
              className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm focus:ring-2 focus:ring-pink-400 focus:outline-none"
            />
          </div>

          <div>
            <label className="block text-xs font-semibold text-slate-700 mb-1.5">
              Warna Identitas
            </label>
            <div className="flex flex-wrap gap-2">
              {COURSE_COLORS.map((c) => (
                <button
                  key={c.hex}
                  type="button"
                  onClick={() => setColor(c.hex)}
                  className={`w-8 h-8 rounded-full flex items-center justify-center transition ${
                    color === c.hex ? 'ring-2 ring-offset-2 ring-slate-800 scale-110' : ''
                  }`}
                  style={{ backgroundColor: c.hex }}
                  title={c.name}
                >
                  {color === c.hex && <Check className="w-4 h-4 text-white" />}
                </button>
              ))}
            </div>
          </div>

          <div className="flex justify-end gap-2 pt-3 border-t border-slate-100">
            <button
              type="button"
              onClick={onClose}
              className="px-4 py-2 rounded-xl text-xs font-semibold text-slate-600 hover:bg-slate-100"
            >
              Batal
            </button>
            <button
              type="submit"
              disabled={saving}
              className="px-5 py-2 rounded-xl bg-pink-500 hover:bg-pink-600 text-white text-xs font-semibold shadow-sm"
            >
              {saving ? 'Menyimpan...' : 'Simpan Mata Kuliah'}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}

function CourseDetailPage({
  courseId,
  courses,
  weeks,
  tasks,
  groupMembers,
  activeGroupId,
  authUser,
  onBack,
  onOpenTask,
  logActivity,
  triggerNotification,
  showToast
}) {
  const course = courses.find((c) => c.id === courseId);
  const courseWeeks = weeks.filter((w) => w.courseId === courseId);
  const courseTasks = tasks.filter((t) => t.courseId === courseId);

  const [weekModal, setWeekModal] = useState(null); // null | 'new' | weekObj
  const [taskModalDefaults, setTaskModalDefaults] = useState(null); // null | { courseId, weekId }
  const [confirmDeleteWeek, setConfirmDeleteWeek] = useState(null);

  if (!course) {
    return (
      <div className="text-center py-12">
        <p className="text-slate-500">Mata kuliah tidak ditemukan.</p>
        <button onClick={onBack} className="mt-3 text-pink-600 font-semibold text-sm">
          Kembali ke Mata Kuliah
        </button>
      </div>
    );
  }

  const completedCount = courseTasks.filter((t) => t.status === 'Selesai').length;
  const incompleteCount = courseTasks.length - completedCount;
  const courseProgress =
    courseTasks.length === 0
      ? 0
      : Math.round(
          courseTasks.reduce((sum, t) => sum + computeTaskProgress(t), 0) / courseTasks.length
        );

  const handleDeleteWeek = async (week) => {
    try {
      await deleteDoc(
        doc(db, 'artifacts', appId, 'public', 'data', `grp_${activeGroupId}_weeks`, week.id)
      );
      await logActivity(`menghapus "${week.title}" pada mata kuliah ${course.name}`, 'week');
      showToast('Week berhasil dihapus.');
      setConfirmDeleteWeek(null);
    } catch {
      showToast('Gagal menghapus Week.', 'error');
    }
  };

  return (
    <div className="space-y-6">
      {/* Breadcrumb Back */}
      <button
        onClick={onBack}
        className="inline-flex items-center gap-1.5 text-xs font-bold text-slate-600 hover:text-pink-600 transition"
      >
        <ChevronLeft className="w-4 h-4" /> Kembali ke Daftar Mata Kuliah
      </button>

      {/* Top Course Summary Banner */}
      <div className="bg-white rounded-3xl border border-slate-100 shadow-xs overflow-hidden">
        <div className="h-2.5 w-full" style={{ backgroundColor: course.color || '#EC4899' }} />
        <div className="p-6">
          <div className="flex flex-col lg:flex-row lg:items-center justify-between gap-4">
            <div>
              <span className="text-xs font-bold px-3 py-1 rounded-full bg-pink-50 text-pink-700">
                {course.semester} • {course.academicYear}
              </span>
              <h1 className="text-2xl sm:text-3xl font-extrabold text-slate-900 mt-2">
                {course.name}
              </h1>
              <p className="text-sm font-medium text-slate-500 mt-0.5">
                Dosen Pengampu: <span className="text-slate-700">{course.lecturer || '-'}</span>
              </p>
              {course.description && (
                <p className="text-xs text-slate-600 mt-2 max-w-2xl">{course.description}</p>
              )}
            </div>

            <div className="flex flex-wrap gap-2">
              <button
                onClick={() => setWeekModal('new')}
                className="px-4 py-2.5 rounded-2xl bg-pink-50 hover:bg-pink-100 text-pink-700 text-xs font-bold flex items-center gap-1.5 transition"
              >
                <Plus className="w-4 h-4" /> Tambah Week
              </button>
              <button
                onClick={() =>
                  setTaskModalDefaults({
                    courseId: course.id,
                    weekId: courseWeeks[0]?.id || ''
                  })
                }
                className="px-4 py-2.5 rounded-2xl bg-gradient-to-r from-pink-500 to-rose-600 text-white text-xs font-bold shadow-md shadow-pink-200 flex items-center gap-1.5 transition"
              >
                <Plus className="w-4 h-4" /> Tambah Tugas
              </button>
            </div>
          </div>

          {/* Course Progress Metrics */}
          <div className="grid grid-cols-2 sm:grid-cols-4 gap-3 mt-6 pt-5 border-t border-slate-100">
            <div className="bg-slate-50 rounded-2xl p-3.5">
              <span className="text-[11px] text-slate-500 block">Progres Mata Kuliah</span>
              <span className="text-xl font-extrabold text-pink-600">{courseProgress}%</span>
              <ProgressBar value={courseProgress} heightClass="h-1.5" />
            </div>
            <div className="bg-slate-50 rounded-2xl p-3.5">
              <span className="text-[11px] text-slate-500 block">Total Tugas</span>
              <span className="text-xl font-extrabold text-slate-900">{courseTasks.length}</span>
            </div>
            <div className="bg-emerald-50/70 rounded-2xl p-3.5">
              <span className="text-[11px] text-emerald-700 block">Tugas Selesai</span>
              <span className="text-xl font-extrabold text-emerald-700">{completedCount}</span>
            </div>
            <div className="bg-amber-50/70 rounded-2xl p-3.5">
              <span className="text-[11px] text-amber-700 block">Belum Selesai</span>
              <span className="text-xl font-extrabold text-amber-700">{incompleteCount}</span>
            </div>
          </div>
        </div>
      </div>

      {/* Flexible Week Management Section */}
      <div className="space-y-4">
        <div className="flex items-center justify-between">
          <h2 className="text-lg font-bold text-slate-900">
            Daftar Minggu Pengerjaan ({courseWeeks.length} Week)
          </h2>
          <button
            onClick={() => setWeekModal('new')}
            className="text-xs font-bold text-pink-600 hover:underline flex items-center gap-1"
          >
            <Plus className="w-3.5 h-3.5" /> Tambah Week Baru
          </button>
        </div>

        {courseWeeks.length === 0 ? (
          <div className="bg-white rounded-3xl border border-dashed border-slate-200 p-10 text-center">
            <p className="text-sm font-bold text-slate-700">Belum ada Week pada mata kuliah ini</p>
            <p className="text-xs text-slate-500 mt-1 mb-4">
              Buat Week 1, Week 2, dan seterusnya untuk mengorganisir tugas secara bertahap.
            </p>
            <button
              onClick={() => setWeekModal('new')}
              className="px-4 py-2 rounded-xl bg-pink-500 text-white text-xs font-semibold"
            >
              + Tambah Week 1
            </button>
          </div>
        ) : (
          courseWeeks.map((week) => {
            const wTasks = courseTasks.filter((t) => t.weekId === week.id);
            const wProgress =
              wTasks.length === 0
                ? 0
                : Math.round(
                    wTasks.reduce((sum, t) => sum + computeTaskProgress(t), 0) / wTasks.length
                  );

            return (
              <div
                key={week.id}
                className="bg-white rounded-3xl border border-slate-100 shadow-xs p-5"
              >
                <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3 pb-4 border-b border-slate-100">
                  <div>
                    <div className="flex items-center gap-2">
                      <h3 className="font-bold text-slate-900 text-base">{week.title}</h3>
                      <span className="text-xs font-bold px-2.5 py-0.5 rounded-full bg-pink-50 text-pink-700">
                        {wProgress}%
                      </span>
                    </div>
                    {week.description && (
                      <p className="text-xs text-slate-500 mt-0.5">{week.description}</p>
                    )}
                  </div>

                  <div className="flex items-center gap-2">
                    <button
                      onClick={() =>
                        setTaskModalDefaults({ courseId: course.id, weekId: week.id })
                      }
                      className="px-3 py-1.5 rounded-xl bg-pink-500 hover:bg-pink-600 text-white text-xs font-semibold flex items-center gap-1"
                    >
                      <Plus className="w-3.5 h-3.5" /> Tambah Tugas
                    </button>
                    <button
                      onClick={() => setWeekModal(week)}
                      className="p-2 rounded-xl text-slate-400 hover:text-pink-600 hover:bg-pink-50"
                      title="Edit Week"
                    >
                      <Edit3 className="w-4 h-4" />
                    </button>
                    <button
                      onClick={() => setConfirmDeleteWeek(week)}
                      className="p-2 rounded-xl text-slate-400 hover:text-red-600 hover:bg-red-50"
                      title="Hapus Week"
                    >
                      <Trash2 className="w-4 h-4" />
                    </button>
                  </div>
                </div>

                {/* Week Progress Bar */}
                <div className="my-3">
                  <ProgressBar value={wProgress} heightClass="h-1.5" />
                </div>

                {/* Tasks inside this Week */}
                {wTasks.length === 0 ? (
                  <p className="text-xs text-slate-400 py-4 text-center">
                    Belum ada tugas pada minggu ini. Klik "+ Tambah Tugas" untuk menambahkan.
                  </p>
                ) : (
                  <div className="grid grid-cols-1 md:grid-cols-2 gap-3 mt-3">
                    {wTasks.map((task) => {
                      const tProg = computeTaskProgress(task);
                      const assigned = groupMembers.filter((m) =>
                        (task.assignedMemberIds || []).includes(m.uid)
                      );
                      return (
                        <div
                          key={task.id}
                          onClick={() => onOpenTask(task.id)}
                          className="p-4 rounded-2xl border border-slate-100 hover:border-pink-300 hover:bg-pink-50/20 cursor-pointer transition flex flex-col justify-between"
                        >
                          <div>
                            <div className="flex items-center justify-between gap-2 mb-2">
                              <span
                                className={`text-[10px] font-bold px-2.5 py-0.5 rounded-full border ${getStatusBadgeStyle(
                                  task.status
                                )}`}
                              >
                                {task.status}
                              </span>
                              <span
                                className={`text-[10px] font-bold px-2 py-0.5 rounded-full border ${getPriorityBadgeStyle(
                                  task.priority
                                )}`}
                              >
                                {task.priority}
                              </span>
                            </div>
                            <h4 className="font-bold text-slate-900 text-sm">{task.title}</h4>
                            {task.description && (
                              <p className="text-xs text-slate-500 line-clamp-1 mt-0.5">
                                {task.description}
                              </p>
                            )}
                          </div>

                          <div className="mt-3 pt-3 border-t border-slate-100">
                            <div className="flex items-center justify-between text-xs mb-1.5">
                              <span className="text-slate-400">
                                Deadline: {formatDateID(task.deadline)}
                              </span>
                              <span className="font-bold text-pink-600">{tProg}%</span>
                            </div>
                            <ProgressBar value={tProg} heightClass="h-1.5" />

                            <div className="flex items-center justify-between mt-3">
                              <div className="flex -space-x-1.5">
                                {assigned.length === 0 ? (
                                  <span className="text-[11px] text-slate-400">
                                    Belum ada anggota
                                  </span>
                                ) : (
                                  assigned.map((m) => (
                                    <AvatarBadge
                                      key={m.uid}
                                      name={m.name}
                                      color={m.avatarColor}
                                      url={m.avatarUrl}
                                      size="xs"
                                    />
                                  ))
                                )}
                              </div>
                              <span className="text-[11px] font-semibold text-pink-600 flex items-center gap-0.5">
                                Progres Anggota <ChevronRight className="w-3.5 h-3.5" />
                              </span>
                            </div>
                          </div>
                        </div>
                      );
                    })}
                  </div>
                )}
              </div>
            );
          })
        )}
      </div>

      {/* Week Add/Edit Modal */}
      {weekModal && (
        <WeekFormModal
          existingWeek={weekModal === 'new' ? null : weekModal}
          course={course}
          nextOrder={courseWeeks.length + 1}
          activeGroupId={activeGroupId}
          onClose={() => setWeekModal(null)}
          logActivity={logActivity}
          showToast={showToast}
        />
      )}

      {/* Task Add Modal */}
      {taskModalDefaults && (
        <TaskFormModal
          defaults={taskModalDefaults}
          courses={courses}
          weeks={weeks}
          groupMembers={groupMembers}
          activeGroupId={activeGroupId}
          authUser={authUser}
          onClose={() => setTaskModalDefaults(null)}
          logActivity={logActivity}
          triggerNotification={triggerNotification}
          showToast={showToast}
        />
      )}

      {confirmDeleteWeek && (
        <ConfirmDialog
          title="Hapus Week?"
          message={`Anda yakin ingin menghapus "${confirmDeleteWeek.title}"?`}
          confirmText="Ya, Hapus"
          onConfirm={() => handleDeleteWeek(confirmDeleteWeek)}
          onCancel={() => setConfirmDeleteWeek(null)}
        />
      )}
    </div>
  );
}

function WeekFormModal({
  existingWeek,
  course,
  nextOrder,
  activeGroupId,
  onClose,
  logActivity,
  showToast
}) {
  const [title, setTitle] = useState(existingWeek?.title || `Week ${nextOrder}: `);
  const [description, setDescription] = useState(existingWeek?.description || '');
  const [saving, setSaving] = useState(false);

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (!title.trim()) return;
    setSaving(true);
    try {
      if (existingWeek) {
        await updateDoc(
          doc(
            db,
            'artifacts',
            appId,
            'public',
            'data',
            `grp_${activeGroupId}_weeks`,
            existingWeek.id
          ),
          {
            title: title.trim(),
            description: description.trim()
          }
        );
        await logActivity(`memperbarui "${title.trim()}" pada ${course.name}`, 'week');
        showToast('Week berhasil diperbarui.');
      } else {
        await addDoc(
          collection(db, 'artifacts', appId, 'public', 'data', `grp_${activeGroupId}_weeks`),
          {
            courseId: course.id,
            title: title.trim(),
            description: description.trim(),
            order: nextOrder,
            createdAt: new Date().toISOString()
          }
        );
        await logActivity(`menambahkan "${title.trim()}" pada ${course.name}`, 'week');
        showToast('Week baru berhasil ditambahkan!');
      }
      onClose();
    } catch {
      showToast('Gagal menyimpan Week.', 'error');
    } finally {
      setSaving(false);
    }
  };

  return (
    <div className="fixed inset-0 z-50 bg-slate-900/50 backdrop-blur-xs flex items-center justify-center p-4">
      <div className="bg-white rounded-3xl max-w-md w-full p-6 shadow-2xl">
        <div className="flex items-center justify-between mb-4">
          <h2 className="text-lg font-bold text-slate-900">
            {existingWeek ? 'Edit Week' : 'Tambah Week Baru'}
          </h2>
          <button onClick={onClose} className="p-1.5 rounded-xl text-slate-400 hover:bg-slate-100">
            <X className="w-5 h-5" />
          </button>
        </div>

        <form onSubmit={handleSubmit} className="space-y-4">
          <div>
            <label className="block text-xs font-semibold text-slate-700 mb-1">
              Nama / Judul Week *
            </label>
            <input
              type="text"
              required
              value={title}
              onChange={(e) => setTitle(e.target.value)}
              placeholder="Contoh: Week 2: Analisis Bisnis & Valuasi"
              className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm focus:ring-2 focus:ring-pink-400 focus:outline-none"
            />
          </div>
          <div>
            <label className="block text-xs font-semibold text-slate-700 mb-1">
              Keterangan / Target Minggu Ini
            </label>
            <textarea
              rows={3}
              value={description}
              onChange={(e) => setDescription(e.target.value)}
              placeholder="Target output pada minggu ini..."
              className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm focus:ring-2 focus:ring-pink-400 focus:outline-none"
            />
          </div>
          <div className="flex justify-end gap-2 pt-2">
            <button
              type="button"
              onClick={onClose}
              className="px-4 py-2 rounded-xl text-xs font-semibold text-slate-600 hover:bg-slate-100"
            >
              Batal
            </button>
            <button
              type="submit"
              disabled={saving}
              className="px-5 py-2 rounded-xl bg-pink-500 hover:bg-pink-600 text-white text-xs font-semibold"
            >
              {saving ? 'Menyimpan...' : 'Simpan Week'}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}

function TaskFormModal({
  existingTask = null,
  defaults = {},
  courses,
  weeks,
  groupMembers,
  activeGroupId,
  authUser,
  onClose,
  logActivity,
  triggerNotification,
  showToast
}) {
  const [title, setTitle] = useState(existingTask?.title || '');
  const [description, setDescription] = useState(existingTask?.description || '');
  const [courseId, setCourseId] = useState(
    existingTask?.courseId || defaults.courseId || courses[0]?.id || ''
  );
  const availableWeeks = useMemo(
    () => weeks.filter((w) => w.courseId === courseId),
    [weeks, courseId]
  );
  const [weekId, setWeekId] = useState(
    existingTask?.weekId || defaults.weekId || availableWeeks[0]?.id || ''
  );
  const [startDate, setStartDate] = useState(
    existingTask?.startDate || new Date().toISOString().split('T')[0]
  );
  const [deadline, setDeadline] = useState(
    existingTask?.deadline || new Date(Date.now() + 7 * 86400000).toISOString().split('T')[0]
  );
  const [priority, setPriority] = useState(existingTask?.priority || 'Sedang');
  const [status, setStatus] = useState(existingTask?.status || 'Belum Dikerjakan');
  const [referenceLink, setReferenceLink] = useState(existingTask?.referenceLink || '');
  const [attachmentName, setAttachmentName] = useState(existingTask?.attachmentName || '');
  const [assignedMemberIds, setAssignedMemberIds] = useState(
    existingTask?.assignedMemberIds || (authUser ? [authUser.uid] : [])
  );
  const [saving, setSaving] = useState(false);

  const toggleMember = (uid) => {
    setAssignedMemberIds((prev) =>
      prev.includes(uid) ? prev.filter((id) => id !== uid) : [...prev, uid]
    );
  };

  const handleFileChange = (e) => {
    const file = e.target.files?.[0];
    if (file) {
      setAttachmentName(file.name);
    }
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (!title.trim()) return;
    if (!courseId) {
      showToast('Pilih mata kuliah terlebih dahulu (buat mata kuliah jika belum ada).', 'error');
      return;
    }
    setSaving(true);
    try {
      // Ensure each assigned member has an initialized memberWork entry
      const existingWork = existingTask?.memberWork || {};
      const updatedMemberWork = { ...existingWork };
      assignedMemberIds.forEach((uid) => {
        if (!updatedMemberWork[uid]) {
          updatedMemberWork[uid] = {
            responsibility: 'Menyelesaikan bagian tugas sesuai pembagian tim',
            doneNotes: '',
            todoNotes: 'Mulai pengerjaan tugas',
            blockers: '',
            workStatus: 'Belum Mulai',
            manualProgress: 0,
            checklists: [],
            updatedAt: new Date().toISOString(),
            updatedByName: ''
          };
        }
      });

      const payload = {
        title: title.trim(),
        description: description.trim(),
        courseId,
        weekId: weekId || availableWeeks[0]?.id || '',
        startDate,
        deadline,
        priority,
        status,
        referenceLink: referenceLink.trim(),
        attachmentName: attachmentName.trim(),
        assignedMemberIds,
        memberWork: updatedMemberWork
      };

      if (existingTask) {
        await updateDoc(
          doc(
            db,
            'artifacts',
            appId,
            'public',
            'data',
            `grp_${activeGroupId}_tasks`,
            existingTask.id
          ),
          payload
        );
        if (existingTask.deadline !== deadline) {
          await logActivity(
            `mengubah deadline tugas "${payload.title}" menjadi ${formatDateID(deadline)}`,
            'task',
            existingTask.id
          );
          await triggerNotification({
            title: 'Perubahan Deadline Tugas',
            message: `Deadline tugas "${payload.title}" diubah menjadi ${formatDateID(deadline)}.`,
            type: 'deadline',
            taskId: existingTask.id
          });
        } else {
          await logActivity(`memperbarui detail tugas "${payload.title}"`, 'task', existingTask.id);
        }
        showToast('Tugas berhasil diperbarui.');
      } else {
        const ref = await addDoc(
          collection(db, 'artifacts', appId, 'public', 'data', `grp_${activeGroupId}_tasks`),
          {
            ...payload,
            revisions: [],
            comments: [],
            createdAt: new Date().toISOString()
          }
        );
        await logActivity(`menambahkan tugas baru "${payload.title}"`, 'task', ref.id);
        await triggerNotification({
          title: 'Tugas Baru Ditambahkan',
          message: `Tugas "${payload.title}" telah dibuat (Deadline: ${formatDateID(deadline)}).`,
          type: 'task',
          targetUserIds: assignedMemberIds,
          taskId: ref.id
        });
        showToast('Tugas baru berhasil dibuat!');
      }
      onClose();
    } catch (err) {
      console.error(err);
      showToast('Gagal menyimpan tugas.', 'error');
    } finally {
      setSaving(false);
    }
  };

  return (
    <div className="fixed inset-0 z-50 bg-slate-900/50 backdrop-blur-xs flex items-center justify-center p-4">
      <div className="bg-white rounded-3xl max-w-2xl w-full p-6 shadow-2xl max-h-[92vh] overflow-y-auto">
        <div className="flex items-center justify-between mb-5">
          <h2 className="text-lg font-bold text-slate-900">
            {existingTask ? 'Edit Tugas' : 'Tambah Tugas Baru'}
          </h2>
          <button onClick={onClose} className="p-1.5 rounded-xl text-slate-400 hover:bg-slate-100">
            <X className="w-5 h-5" />
          </button>
        </div>

        <form onSubmit={handleSubmit} className="space-y-4">
          <div>
            <label className="block text-xs font-semibold text-slate-700 mb-1">Nama Tugas *</label>
            <input
              type="text"
              required
              value={title}
              onChange={(e) => setTitle(e.target.value)}
              placeholder="Contoh: Pembuatan PPT Presentasi Valuasi"
              className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm focus:ring-2 focus:ring-pink-400 focus:outline-none"
            />
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
            <div>
              <label className="block text-xs font-semibold text-slate-700 mb-1">
                Mata Kuliah *
              </label>
              <select
                required
                value={courseId}
                onChange={(e) => {
                  const newCid = e.target.value;
                  setCourseId(newCid);
                  const firstW = weeks.find((w) => w.courseId === newCid);
                  setWeekId(firstW?.id || '');
                }}
                className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm focus:ring-2 focus:ring-pink-400 focus:outline-none"
              >
                <option value="">-- Pilih Mata Kuliah --</option>
                {courses.map((c) => (
                  <option key={c.id} value={c.id}>
                    {c.name}
                  </option>
                ))}
              </select>
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-700 mb-1">
                Week Pengerjaan
              </label>
              <select
                value={weekId}
                onChange={(e) => setWeekId(e.target.value)}
                className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm focus:ring-2 focus:ring-pink-400 focus:outline-none"
              >
                {availableWeeks.length === 0 && <option value="">Tanpa Week</option>}
                {availableWeeks.map((w) => (
                  <option key={w.id} value={w.id}>
                    {w.title}
                  </option>
                ))}
              </select>
            </div>
          </div>

          <div>
            <label className="block text-xs font-semibold text-slate-700 mb-1">
              Deskripsi Tugas
            </label>
            <textarea
              rows={2}
              value={description}
              onChange={(e) => setDescription(e.target.value)}
              placeholder="Instruksi pengerjaan, kriteria penilaian, atau ruang lingkup tugas..."
              className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm focus:ring-2 focus:ring-pink-400 focus:outline-none"
            />
          </div>

          <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
            <div>
              <label className="block text-xs font-semibold text-slate-700 mb-1">
                Tanggal Mulai
              </label>
              <input
                type="date"
                value={startDate}
                onChange={(e) => setStartDate(e.target.value)}
                className="w-full px-3 py-2 rounded-xl border border-slate-200 text-xs focus:ring-2 focus:ring-pink-400 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-700 mb-1">Deadline *</label>
              <input
                type="date"
                required
                value={deadline}
                onChange={(e) => setDeadline(e.target.value)}
                className="w-full px-3 py-2 rounded-xl border border-slate-200 text-xs focus:ring-2 focus:ring-pink-400 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-700 mb-1">Prioritas</label>
              <select
                value={priority}
                onChange={(e) => setPriority(e.target.value)}
                className="w-full px-3 py-2 rounded-xl border border-slate-200 text-xs focus:ring-2 focus:ring-pink-400 focus:outline-none"
              >
                {PRIORITY_OPTIONS.map((p) => (
                  <option key={p} value={p}>
                    {p}
                  </option>
                ))}
              </select>
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-700 mb-1">Status</label>

                    {s}
                  </option>
                ))}
              </select>

            <label className="block text-xs font-semibold text-slate-700 mb-1.5">
              Bagikan Tugas ke Anggota Kelompok (Dapat pilih lebih dari 1)
            </label>
            <div className="flex flex-wrap gap-2">
              {groupMembers.map((m) => {
                const selected = assignedMemberIds.includes(m.uid);
                return (
                  <button
                    key={m.uid}
                    type="button"
                    onClick={() => toggleMember(m.uid)}
                    className={`flex items-center gap-2 px-3 py-1.5 rounded-xl border text-xs font-semibold transition ${
                      selected
                        ? 'bg-pink-50 border-pink-400 text-pink-700'
                        : 'bg-white border-slate-200 text-slate-600 hover:bg-slate-50'
                    }`}
                  >
                    <AvatarBadge name={m.name} color={m.avatarColor} url={m.avatarUrl} size="xs" />
                    <span>{m.name}</span>
                    {selected && <Check className="w-3.5 h-3.5 text-pink-600" />}
                  </button>
                );
              })}
            </div>
          </div>

          {/* Reference Link & File Attachment */}
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
            <div>
              <label className="block text-xs font-semibold text-slate-700 mb-1">
                Link Referensi (Google Docs / Figma / Drive)
              </label>
              <input
                type="url"
                value={referenceLink}
                onChange={(e) => setReferenceLink(e.target.value)}
                placeholder="https://..."
                className="w-full px-3.5 py-2 rounded-xl border border-slate-200 text-xs focus:ring-2 focus:ring-pink-400 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-700 mb-1">
                Lampiran File / Dokumen
              </label>
              <div className="flex items-center gap-2">
                <label className="flex-1 px-3.5 py-2 rounded-xl border border-dashed border-slate-300 hover:border-pink-400 text-xs text-slate-600 cursor-pointer truncate flex items-center gap-2">
                  <Paperclip className="w-3.5 h-3.5 text-pink-500 shrink-0" />
                  <span className="truncate">
                    {attachmentName || 'Pilih file lampiran...'}
                  </span>
                  <input type="file" onChange={handleFileChange} className="hidden" />
                </label>
                {attachmentName && (
                  <button
                    type="button"
                    onClick={() => setAttachmentName('')}
                    className="p-2 text-slate-400 hover:text-red-500"
                  >
                    <X className="w-4 h-4" />
                  </button>
                )}
              </div>
            </div>
          </div>

          <div className="flex justify-end gap-2 pt-3 border-t border-slate-100">
            <button
              type="button"
              onClick={onClose}
              className="px-4 py-2 rounded-xl text-xs font-semibold text-slate-600 hover:bg-slate-100"
            >
              Batal
            </button>
            <button
              type="submit"
              disabled={saving}
              className="px-5 py-2 rounded-xl bg-pink-500 hover:bg-pink-600 text-white text-xs font-semibold shadow-sm"
            >
              {saving ? 'Menyimpan...' : existingTask ? 'Simpan Perubahan' : 'Buat Tugas'}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}

// ============================================================================
// CORE FEATURE: TASK DETAIL MODAL (PROGRES ANGGOTA, CHECKLIST, REVISI, DISKUSI)
// ============================================================================
function TaskDetailModal({
  task,
  courses,
  weeks,
  groupMembers,
  authUser,
  userProfile,
  isLeader,
  activeGroupId,
  onClose,
  logActivity,
  triggerNotification,
  showToast
}) {
  const [activeTab, setActiveTab] = useState('members'); // 'members' | 'revisions' | 'comments'
  const [showEditTask, setShowEditTask] = useState(false);
  const [confirmDeleteTask, setConfirmDeleteTask] = useState(false);

  const course = courses.find((c) => c.id === task.courseId);
  const courseWeeks = weeks.filter((w) => w.courseId === task.courseId);
  const taskProgress = computeTaskProgress(task);

  const assignedMembers = useMemo(() => {
    return (task.assignedMemberIds || []).map((uid) => {
      const m = groupMembers.find((gm) => gm.uid === uid);
      return (
        m || {
          uid,
          name: 'Anggota Tim',
          avatarColor: 'bg-slate-400',
          role: 'Anggota Kelompok'
        }
      );
    });
  }, [task.assignedMemberIds, groupMembers]);

  // Check if all assigned members reached 100%
  const allMembersCompleted =
    assignedMembers.length > 0 &&
    assignedMembers.every((m) => computeMemberProgress(task.memberWork?.[m.uid]) === 100);

  const taskDocRef = doc(
    db,
    'artifacts',
    appId,
    'public',
    'data',
    `grp_${activeGroupId}_tasks`,
    task.id
  );

  // Update overall task status
  const handleStatusChange = async (newStatus) => {
    try {
      await updateDoc(taskDocRef, { status: newStatus });
      await logActivity(`mengubah status tugas "${task.title}" menjadi ${newStatus}`, 'task', task.id);
      await triggerNotification({
        title: 'Status Tugas Diperbarui',
        message: `Tugas "${task.title}" kini berstatus: ${newStatus}.`,
        type: 'status',
        taskId: task.id
      });
      showToast(`Status tugas diubah menjadi "${newStatus}".`);
    } catch {
      showToast('Gagal mengubah status tugas.', 'error');
    }
  };

  // Move task to another Week
  const handleMoveWeek = async (newWeekId) => {
    try {
      await updateDoc(taskDocRef, { weekId: newWeekId });
      const targetWeek = weeks.find((w) => w.id === newWeekId);
      await logActivity(
        `memindahkan tugas "${task.title}" ke ${targetWeek?.title || 'Week lain'}`,
        'task',
        task.id
      );
      showToast('Tugas berhasil dipindahkan ke Week baru.');
    } catch {
      showToast('Gagal memindahkan Week.', 'error');
    }
  };

  // Add an unassigned member directly from Task Detail
  const handleAddMemberToTask = async (uid) => {
    if (!uid || (task.assignedMemberIds || []).includes(uid)) return;
    try {
      const nextIds = [...(task.assignedMemberIds || []), uid];
      const nextWork = { ...(task.memberWork || {}) };
      if (!nextWork[uid]) {
        nextWork[uid] = {
          responsibility: 'Menyelesaikan bagian pekerjaan yang ditugaskan',
          doneNotes: '',
          todoNotes: '',
          blockers: '',
          workStatus: 'Belum Mulai',
          manualProgress: 0,
          checklists: [],
          updatedAt: new Date().toISOString(),
          updatedByName: userProfile?.name || ''
        };
      }
      await updateDoc(taskDocRef, {
        assignedMemberIds: nextIds,
        memberWork: nextWork
      });
      const addedM = groupMembers.find((m) => m.uid === uid);
      await logActivity(
        `menambahkan ${addedM?.name || 'anggota'} ke tugas "${task.title}"`,
        'task',
        task.id
      );
      await triggerNotification({
        title: 'Pembagian Tugas Baru',
        message: `Anda ditugaskan pada "${task.title}".`,
        type: 'assignment',
        targetUserIds: [uid],
        taskId: task.id
      });
      showToast(`${addedM?.name || 'Anggota'} ditambahkan ke tugas ini.`);
    } catch {
      showToast('Gagal menambahkan anggota.', 'error');
    }
  };

  const handleDeleteTask = async () => {
    try {
      await deleteDoc(taskDocRef);
      await logActivity(`menghapus tugas "${task.title}"`, 'task');
      showToast('Tugas berhasil dihapus.');
      onClose();
    } catch {
      showToast('Gagal menghapus tugas.', 'error');
    }
  };

  const unassignedGroupMembers = groupMembers.filter(
    (m) => !(task.assignedMemberIds || []).includes(m.uid)
  );

  return (
    <div className="fixed inset-0 z-50 bg-slate-900/60 backdrop-blur-xs flex items-center justify-center p-2 sm:p-4">
      <div className="bg-slate-50 rounded-3xl max-w-5xl w-full max-h-[94vh] flex flex-col shadow-2xl border border-slate-200 overflow-hidden">
        {/* Top Modal Header */}
        <div className="bg-white px-5 sm:px-6 py-4 border-b border-slate-200 flex items-start justify-between gap-4">
          <div className="min-w-0">
            <div className="flex flex-wrap items-center gap-2 mb-1">
              <span className="text-xs font-bold px-2.5 py-0.5 rounded-full bg-pink-50 text-pink-700">
                {course?.name || 'Mata Kuliah'}
              </span>
              <span
                className={`text-xs font-bold px-2.5 py-0.5 rounded-full border ${getPriorityBadgeStyle(
                  task.priority
                )}`}
              >
                Prioritas {task.priority}
              </span>
              <span className="text-xs text-slate-400">
                Deadline: <strong className="text-slate-700">{formatDateID(task.deadline)}</strong>
              </span>
            </div>
            <h2 className="text-lg sm:text-xl font-extrabold text-slate-900 truncate">
              {task.title}
            </h2>
          </div>

          <div className="flex items-center gap-1.5 shrink-0">
            <button
              onClick={() => setShowEditTask(true)}
              className="p-2 rounded-xl text-slate-500 hover:text-pink-600 hover:bg-pink-50 transition"
              title="Edit Detail Tugas"
            >
              <Edit3 className="w-4 h-4" />
            </button>
            <button
              onClick={() => setConfirmDeleteTask(true)}
              className="p-2 rounded-xl text-slate-500 hover:text-red-600 hover:bg-red-50 transition"
              title="Hapus Tugas"
            >
              <Trash2 className="w-4 h-4" />
            </button>
            <button
              onClick={onClose}
              className="p-2 rounded-xl text-slate-500 hover:bg-slate-100 transition"
            >
              <X className="w-5 h-5" />
            </button>
          </div>
        </div>

        {/* Task Control Bar & Overall Progress */}
        <div className="bg-white px-5 sm:px-6 py-3.5 border-b border-slate-200/80 flex flex-col lg:flex-row lg:items-center justify-between gap-4">
          <div className="flex-1 max-w-md">
            <div className="flex items-center justify-between text-xs mb-1">
              <span className="font-semibold text-slate-600">Progres Tugas Keseluruhan</span>
              <span className="font-extrabold text-pink-600">{taskProgress}%</span>
            </div>
            <ProgressBar value={taskProgress} heightClass="h-2.5" />
          </div>

          <div className="flex flex-wrap items-center gap-2.5">
            {/* Move Week Dropdown */}
            {courseWeeks.length > 0 && (
              <select
                value={task.weekId || ''}
                onChange={(e) => handleMoveWeek(e.target.value)}
                className="px-3 py-1.5 rounded-xl border border-slate-200 bg-slate-50 text-xs font-semibold text-slate-700 focus:outline-none focus:ring-2 focus:ring-pink-400"
                title="Pindahkan ke Week lain"
              >
                {courseWeeks.map((w) => (
                  <option key={w.id} value={w.id}>
                    {w.title}
                  </option>
                ))}
              </select>
            )}

            {/* Status Selector */}
            <select
              value={task.status}
              onChange={(e) => handleStatusChange(e.target.value)}
              className={`px-3 py-1.5 rounded-xl border text-xs font-bold focus:outline-none ${getStatusBadgeStyle(
                task.status
              )}`}
            >
              {STATUS_OPTIONS.map((s) => (
                <option key={s} value={s}>
                  Status: {s}
                </option>
              ))}
            </select>
          </div>
        </div>

        {/* Banner when all members reach 100% */}
        {allMembersCompleted && task.status !== 'Selesai' && (
          <div className="bg-emerald-50 border-b border-emerald-200 px-6 py-2.5 flex flex-wrap items-center justify-between gap-2">
            <div className="flex items-center gap-2 text-xs font-semibold text-emerald-800">
              <CheckCircle2 className="w-4 h-4 text-emerald-600 shrink-0" />
              <span>
                Seluruh anggota telah menyelesaikan 100% bagiannya! Tandai tugas ini selesai atau menunggu review?
              </span>
            </div>
            <div className="flex items-center gap-2">
              <button
                onClick={() => handleStatusChange('Menunggu Review')}
                className="px-3 py-1 rounded-lg bg-white border border-emerald-300 text-emerald-700 text-xs font-bold hover:bg-emerald-100"
              >
                Menunggu Review
              </button>
              <button
                onClick={() => handleStatusChange('Selesai')}
                className="px-3 py-1 rounded-lg bg-emerald-600 text-white text-xs font-bold hover:bg-emerald-700"
              >
                Tandai Selesai
              </button>
            </div>
          </div>
        )}

        {/* Description, Links, Attachments */}
        {(task.description || task.referenceLink || task.attachmentName) && (
          <div className="bg-pink-50/40 px-5 sm:px-6 py-3 border-b border-pink-100/70 flex flex-wrap items-center justify-between gap-3 text-xs">
            <p className="text-slate-700 flex-1 min-w-[200px]">
              {task.description || 'Tidak ada catatan deskripsi.'}
            </p>
            <div className="flex flex-wrap items-center gap-2">
              {task.referenceLink && (
                <a
                  href={task.referenceLink}
                  target="_blank"
                  rel="noreferrer"
                  className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-xl bg-white border border-pink-200 text-pink-700 font-semibold hover:bg-pink-50"
                >
                  <ExternalLink className="w-3.5 h-3.5" /> Link Referensi
                </a>
              )}
              {task.attachmentName && (
                <span className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-xl bg-white border border-slate-200 text-slate-700 font-medium">
                  <Paperclip className="w-3.5 h-3.5 text-pink-500" /> {task.attachmentName}
                </span>
              )}
            </div>
          </div>
        )}

        {/* Section Tabs */}
        <div className="bg-white px-5 sm:px-6 border-b border-slate-200 flex items-center gap-2 overflow-x-auto">
          <button
            onClick={() => setActiveTab('members')}
            className={`py-3 px-3 text-xs sm:text-sm font-bold border-b-2 transition whitespace-nowrap flex items-center gap-2 ${
              activeTab === 'members'
                ? 'border-pink-500 text-pink-600'
                : 'border-transparent text-slate-500 hover:text-slate-800'
            }`}
          >
            <Users className="w-4 h-4" />
            <span>Progres Anggota ({assignedMembers.length})</span>
          </button>

          <button
            onClick={() => setActiveTab('revisions')}
            className={`py-3 px-3 text-xs sm:text-sm font-bold border-b-2 transition whitespace-nowrap flex items-center gap-2 ${
              activeTab === 'revisions'
                ? 'border-pink-500 text-pink-600'
                : 'border-transparent text-slate-500 hover:text-slate-800'
            }`}
          >
            <RotateCcw className="w-4 h-4" />
            <span>Revisi Tugas ({(task.revisions || []).length})</span>
          </button>

          <button
            onClick={() => setActiveTab('comments')}
            className={`py-3 px-3 text-xs sm:text-sm font-bold border-b-2 transition whitespace-nowrap flex items-center gap-2 ${
              activeTab === 'comments'
                ? 'border-pink-500 text-pink-600'
                : 'border-transparent text-slate-500 hover:text-slate-800'
            }`}
          >
            <MessageSquare className="w-4 h-4" />
            <span>Diskusi & Komentar ({(task.comments || []).length})</span>
          </button>
        </div>

        {/* Scrollable Modal Content */}
        <div className="flex-1 overflow-y-auto p-5 sm:p-6">
          {activeTab === 'members' && (
            <div className="space-y-4">
              <div className="flex flex-wrap items-center justify-between gap-3">
                <div>
                  <h3 className="text-base font-bold text-slate-900">
                    Pembagian & Pencatatan Progres Per Anggota
                  </h3>
                  <p className="text-xs text-slate-500">
                    Setiap anggota memiliki kolom mandiri untuk mencatat pekerjaan selesai, belum selesai, kendala, dan checklist.
                  </p>
                </div>

                {unassignedGroupMembers.length > 0 && (
                  <div className="flex items-center gap-2">
                    <select
                      onChange={(e) => {
                        if (e.target.value) {
                          handleAddMemberToTask(e.target.value);
                          e.target.value = '';
                        }
                      }}
                      defaultValue=""
                      className="px-3 py-2 rounded-xl border border-pink-200 bg-white text-xs font-semibold text-pink-700 focus:outline-none"
                    >
                      <option value="" disabled>
                        + Tambah Anggota ke Tugas Ini
                      </option>
                      {unassignedGroupMembers.map((m) => (
                        <option key={m.uid} value={m.uid}>
                          {m.name}
                        </option>
                      ))}
                    </select>
                  </div>
                )}
              </div>

              {assignedMembers.length === 0 ? (
                <div className="bg-white rounded-2xl border border-dashed border-slate-200 p-8 text-center">
                  <p className="text-sm font-bold text-slate-700">
                    Belum ada anggota yang ditugaskan pada tugas ini.
                  </p>
                  <p className="text-xs text-slate-500 mt-1">
                    Gunakan dropdown di atas untuk membagikan tugas kepada anggota kelompok.
                  </p>
                </div>
              ) : (
                <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
                  {assignedMembers.map((member) => (
                    <MemberWorkCard
                      key={member.uid}
                      member={member}
                      task={task}
                      workData={task.memberWork?.[member.uid]}
                      canEdit={member.uid === authUser.uid || isLeader}
                      currentUserName={userProfile?.name || 'Anggota'}
                      taskDocRef={taskDocRef}
                      logActivity={logActivity}
                      triggerNotification={triggerNotification}
                      showToast={showToast}
                    />
                  ))}
                </div>
              )}
            </div>
          )}

          {activeTab === 'revisions' && (
            <TaskRevisionsSection
              task={task}
              groupMembers={groupMembers}
              userProfile={userProfile}
              taskDocRef={taskDocRef}
              logActivity={logActivity}
              triggerNotification={triggerNotification}
              showToast={showToast}
            />
          )}

          {activeTab === 'comments' && (
            <TaskCommentsSection
              task={task}
              groupMembers={groupMembers}
              authUser={authUser}
              userProfile={userProfile}
              taskDocRef={taskDocRef}
              logActivity={logActivity}
              triggerNotification={triggerNotification}
              showToast={showToast}
            />
          )}
        </div>
      </div>

      {showEditTask && (
        <TaskFormModal
          existingTask={task}
          courses={courses}
          weeks={weeks}
          groupMembers={groupMembers}
          activeGroupId={activeGroupId}
          authUser={authUser}
          onClose={() => setShowEditTask(false)}
          logActivity={logActivity}
          triggerNotification={triggerNotification}
          showToast={showToast}
        />
      )}

      {confirmDeleteTask && (
        <ConfirmDialog
          title="Hapus Tugas?"
          message={`Anda yakin ingin menghapus tugas "${task.title}"?`}
          confirmText="Ya, Hapus"
          onConfirm={handleDeleteTask}
          onCancel={() => setConfirmDeleteTask(false)}
        />
      )}
    </div>
  );
}

// ============================================================================
// PER-MEMBER WORK CARD (INSIDE TASK DETAIL MODAL)
// ============================================================================
function MemberWorkCard({
  member,
  task,
  workData,
  canEdit,
  currentUserName,
  taskDocRef,
  logActivity,
  triggerNotification,
  showToast
}) {
  const safeWork = workData || {
    responsibility: '',
    doneNotes: '',
    todoNotes: '',
    blockers: '',
    workStatus: 'Belum Mulai',
    manualProgress: 0,
    checklists: [],
    updatedAt: '',
    updatedByName: ''
  };

  const [editingNotes, setEditingNotes] = useState(false);
  const [responsibility, setResponsibility] = useState(safeWork.responsibility || '');
  const [doneNotes, setDoneNotes] = useState(safeWork.doneNotes || '');
  const [todoNotes, setTodoNotes] = useState(safeWork.todoNotes || '');
  const [blockers, setBlockers] = useState(safeWork.blockers || '');
  const [workStatus, setWorkStatus] = useState(safeWork.workStatus || 'Belum Mulai');
  const [manualProgress, setManualProgress] = useState(safeWork.manualProgress || 0);
  const [newCheckItem, setNewCheckItem] = useState('');

  useEffect(() => {
    setResponsibility(safeWork.responsibility || '');
    setDoneNotes(safeWork.doneNotes || '');
    setTodoNotes(safeWork.todoNotes || '');
    setBlockers(safeWork.blockers || '');
    setWorkStatus(safeWork.workStatus || 'Belum Mulai');
    setManualProgress(safeWork.manualProgress || 0);
  }, [workData]);

  const checklists = safeWork.checklists || [];
  const memberProgress = computeMemberProgress(safeWork);

  const persistMemberWork = async (updatedFields, activityMsg = null, notifyMsg = null) => {
    try {
      const nextMemberWork = {
        ...safeWork,
        ...updatedFields,
        updatedAt: new Date().toISOString(),
        updatedByName: currentUserName
      };

      await updateDoc(taskDocRef, {
        [`memberWork.${member.uid}`]: nextMemberWork
      });

      if (activityMsg) {
        await logActivity(activityMsg, 'progress', task.id);
      }
      if (notifyMsg) {
        await triggerNotification({
          title: 'Pembaruan Progres Anggota',
          message: notifyMsg,
          type: 'progress',
          taskId: task.id
        });
      }
    } catch (err) {
      console.error(err);
      showToast('Gagal menyimpan progres anggota.', 'error');
    }
  };

  const handleSaveNotes = async (e) => {
    e.preventDefault();
    const calcPct =
      checklists.length > 0
        ? Math.round((checklists.filter((c) => c.done).length / checklists.length) * 100)
        : Number(manualProgress);

    await persistMemberWork(
      {
        responsibility: responsibility.trim(),
        doneNotes: doneNotes.trim(),
        todoNotes: todoNotes.trim(),
        blockers: blockers.trim(),
        workStatus,
        manualProgress: Number(manualProgress)
      },
      `memperbarui catatan progres ${member.name} pada tugas "${task.title}" (${calcPct}%)`,
      `${currentUserName} memperbarui progres ${member.name} pada "${task.title}" menjadi ${calcPct}%.`
    );
    setEditingNotes(false);
    showToast(`Progres ${member.name} tersimpan!`);
  };

  const handleAddChecklist = async (e) => {
    e.preventDefault();
    if (!newCheckItem.trim()) return;
    const nextList = [
      ...checklists,
      {
        id: 'chk_' + Date.now(),
        text: newCheckItem.trim(),
        done: false
      }
    ];
    setNewCheckItem('');
    const newPct = Math.round((nextList.filter((c) => c.done).length / nextList.length) * 100);
    await persistMemberWork(
      { checklists: nextList },
      `menambahkan item pekerjaan "${newCheckItem.trim()}" untuk ${member.name} pada "${task.title}" (${newPct}%)`
    );
  };

  const handleToggleChecklist = async (itemId) => {
    const targetItem = checklists.find((c) => c.id === itemId);
    const nextList = checklists.map((c) => (c.id === itemId ? { ...c, done: !c.done } : c));
    const doneCount = nextList.filter((c) => c.done).length;
    const newPct = nextList.length === 0 ? 0 : Math.round((doneCount / nextList.length) * 100);
    const nextStatus =
      newPct === 100 ? 'Selesai' : newPct > 0 ? 'Sedang Dikerjakan' : 'Belum Mulai';

    await persistMemberWork(
      {
        checklists: nextList,
        workStatus: nextStatus
      },
      `${!targetItem?.done ? 'menyelesaikan' : 'membatalkan'} checklist "${targetItem?.text}" (${member.name}) — progres ${newPct}%`,
      !targetItem?.done
        ? `${member.name} menyelesaikan item "${targetItem?.text}" pada "${task.title}" (${newPct}%).`
        : null
    );
  };

  const handleDeleteChecklist = async (itemId) => {
    const nextList = checklists.filter((c) => c.id !== itemId);
    await persistMemberWork({ checklists: nextList });
  };

  return (
    <div className="bg-white rounded-3xl border border-slate-200/80 shadow-xs p-5 flex flex-col justify-between">
      <div>
        {/* Card Header: Member Identity & Individual Progress */}
        <div className="flex items-start justify-between gap-3 pb-3.5 border-b border-slate-100">
          <div className="flex items-center gap-3 min-w-0">
            <AvatarBadge
              name={member.name}
              color={member.avatarColor}
              url={member.avatarUrl}
              size="md"
            />
            <div className="min-w-0">
              <h4 className="font-bold text-slate-900 text-sm truncate">{member.name}</h4>
              <div className="flex flex-wrap items-center gap-1.5 mt-0.5">
                <span
                  className={`text-[10px] font-bold px-2 py-0.5 rounded-full border ${getStatusBadgeStyle(
                    safeWork.workStatus
                  )}`}
                >
                  {safeWork.workStatus || 'Belum Mulai'}
                </span>
                {safeWork.updatedAt && (
                  <span className="text-[10px] text-slate-400">
                    Update: {formatDateTimeID(safeWork.updatedAt)}
                    {safeWork.updatedByName ? ` oleh ${safeWork.updatedByName}` : ''}
                  </span>
                )}
              </div>
            </div>
          </div>

          <div className="text-right shrink-0">
            <span className="text-lg font-extrabold text-pink-600">{memberProgress}%</span>
            {canEdit && !editingNotes && (
              <button
                onClick={() => setEditingNotes(true)}
                className="block text-[11px] font-bold text-pink-600 hover:underline mt-0.5"
              >
                Edit Catatan
              </button>
            )}
          </div>
        </div>

        {/* Individual Progress Bar */}
        <div className="my-3">
          <ProgressBar value={memberProgress} heightClass="h-2" />
        </div>

        {/* Editable Notes Form vs Read View */}
        {editingNotes ? (
          <form onSubmit={handleSaveNotes} className="space-y-3 bg-pink-50/40 p-3.5 rounded-2xl border border-pink-100 my-3">
            <div>
              <label className="block text-[11px] font-bold text-slate-700 mb-1">
                Bagian Pekerjaan / Tanggung Jawab
              </label>
              <textarea
                rows={2}
                value={responsibility}
                onChange={(e) => setResponsibility(e.target.value)}
                placeholder="Contoh: Membuat slide latar belakang & analisis..."
                className="w-full px-3 py-1.5 rounded-xl border border-slate-200 bg-white text-xs focus:ring-2 focus:ring-pink-400 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-[11px] font-bold text-emerald-700 mb-1">
                Sudah Dikerjakan
              </label>
              <textarea
                rows={2}
                value={doneNotes}
                onChange={(e) => setDoneNotes(e.target.value)}
                placeholder="Catat bagian yang telah selesai..."
                className="w-full px-3 py-1.5 rounded-xl border border-slate-200 bg-white text-xs focus:ring-2 focus:ring-pink-400 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-[11px] font-bold text-amber-700 mb-1">
                Belum Dikerjakan
              </label>
              <textarea
                rows={2}
                value={todoNotes}
                onChange={(e) => setTodoNotes(e.target.value)}
                placeholder="Catat sisa pekerjaan yang belum selesai..."
                className="w-full px-3 py-1.5 rounded-xl border border-slate-200 bg-white text-xs focus:ring-2 focus:ring-pink-400 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-[11px] font-bold text-red-600 mb-1">
                Kendala / Hambatan (Opsional)
              </label>
              <input
                type="text"
                value={blockers}
                onChange={(e) => setBlockers(e.target.value)}
                placeholder="Contoh: Menunggu data laporan keuangan..."
                className="w-full px-3 py-1.5 rounded-xl border border-slate-200 bg-white text-xs focus:ring-2 focus:ring-pink-400 focus:outline-none"
              />
            </div>

            <div className="grid grid-cols-2 gap-2">
              <div>
                <label className="block text-[11px] font-bold text-slate-700 mb-1">
                  Status Individu
                </label>
                <select
                  value={workStatus}
                  onChange={(e) => setWorkStatus(e.target.value)}
                  className="w-full px-2.5 py-1.5 rounded-xl border border-slate-200 bg-white text-xs"
                >
                  {MEMBER_WORK_STATUS_OPTIONS.map((st) => (
                    <option key={st} value={st}>
                      {st}
                    </option>
                  ))}
                </select>
              </div>
              {checklists.length === 0 && (
                <div>
                  <label className="block text-[11px] font-bold text-slate-700 mb-1">
                    Progres Manual ({manualProgress}%)
                  </label>
                  <input
                    type="range"
                    min="0"
                    max="100"
                    step="5"
                    value={manualProgress}
                    onChange={(e) => setManualProgress(Number(e.target.value))}
                    className="w-full accent-pink-500 mt-1"
                  />
                </div>
              )}
            </div>

            <div className="flex justify-end gap-2 pt-1">
              <button
                type="button"
                onClick={() => setEditingNotes(false)}
                className="px-3 py-1.5 rounded-xl text-xs font-semibold text-slate-600 hover:bg-white"
              >
                Batal
              </button>
              <button
                type="submit"
                className="px-4 py-1.5 rounded-xl bg-pink-500 text-white text-xs font-semibold"
              >
                Simpan Catatan
              </button>
            </div>
          </form>
        ) : (
          <div className="space-y-2.5 text-xs my-3">
            <div className="bg-slate-50 p-2.5 rounded-xl">
              <span className="font-bold text-slate-700 block text-[11px]">
                Tanggung Jawab:
              </span>
              <p className="text-slate-600 mt-0.5 whitespace-pre-line">
                {safeWork.responsibility || 'Belum diisi'}
              </p>
            </div>

            <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
              <div className="bg-emerald-50/60 border border-emerald-100 p-2.5 rounded-xl">
                <span className="font-bold text-emerald-800 block text-[11px]">
                  Sudah Dikerjakan:
                </span>
                <p className="text-emerald-900/80 mt-0.5 whitespace-pre-line">
                  {safeWork.doneNotes || 'Belum ada catatan'}
                </p>
              </div>
              <div className="bg-amber-50/60 border border-amber-100 p-2.5 rounded-xl">
                <span className="font-bold text-amber-800 block text-[11px]">
                  Belum Dikerjakan:
                </span>
                <p className="text-amber-900/80 mt-0.5 whitespace-pre-line">
                  {safeWork.todoNotes || 'Belum ada catatan'}
                </p>
              </div>
            </div>

            {safeWork.blockers && (
              <div className="bg-red-50/70 border border-red-100 p-2.5 rounded-xl flex items-start gap-2">
                <AlertTriangle className="w-4 h-4 text-red-500 shrink-0 mt-0.5" />
                <div>
                  <span className="font-bold text-red-700 block text-[11px]">Kendala:</span>
                  <p className="text-red-800">{safeWork.blockers}</p>
                </div>
              </div>
            )}
          </div>
        )}

        {/* Interactive Checklist Section */}
        <div className="pt-3 border-t border-slate-100">
          <div className="flex items-center justify-between mb-2">
            <span className="text-xs font-bold text-slate-800">
              Checklist Pekerjaan ({checklists.filter((c) => c.done).length}/{checklists.length})
            </span>
            {checklists.length > 0 && (
              <span className="text-[10px] text-pink-600 font-semibold">Otomatis hitung %</span>
            )}
          </div>

          <div className="space-y-1.5 mb-2.5">
            {checklists.map((item) => (
              <div
                key={item.id}
                className="flex items-center justify-between gap-2 p-2 rounded-xl bg-slate-50 hover:bg-pink-50/40 transition group"
              >
                <button
                  type="button"
                  disabled={!canEdit}
                  onClick={() => handleToggleChecklist(item.id)}
                  className="flex items-center gap-2 text-left flex-1 min-w-0"
                >
                  {item.done ? (
                    <CheckCircle2 className="w-4 h-4 text-emerald-500 shrink-0" />
                  ) : (
                    <Circle className="w-4 h-4 text-slate-400 shrink-0" />
                  )}
                  <span
                    className={`text-xs truncate ${
                      item.done ? 'line-through text-slate-400' : 'text-slate-700 font-medium'
                    }`}
                  >
                    {item.text}
                  </span>
                </button>
                {canEdit && (
                  <button
                    type="button"
                    onClick={() => handleDeleteChecklist(item.id)}
                    className="opacity-0 group-hover:opacity-100 text-slate-400 hover:text-red-500 p-0.5"
                  >
                    <X className="w-3.5 h-3.5" />
                  </button>
                )}
              </div>
            ))}
          </div>

          {canEdit && (
            <form onSubmit={handleAddChecklist} className="flex items-center gap-1.5">
              <input
                type="text"
                value={newCheckItem}
                onChange={(e) => setNewCheckItem(e.target.value)}
                placeholder="+ Tambah pekerjaan baru..."
                className="flex-1 px-3 py-1.5 rounded-xl border border-slate-200 text-xs focus:ring-2 focus:ring-pink-400 focus:outline-none"
              />
              <button
                type="submit"
                className="px-3 py-1.5 rounded-xl bg-pink-500 hover:bg-pink-600 text-white text-xs font-semibold"
              >
                Tambah
              </button>
            </form>
          )}
        </div>
      </div>
    </div>
  );
}

// TASK REVISIONS SECTION (PRESERVES WORK HISTORY)
// ============================================================================
function TaskRevisionsSection({
  task,
  groupMembers,
  userProfile,
  taskDocRef,
  logActivity,
  triggerNotification,
  showToast
}) {
  const [showAddRev, setShowAddRev] = useState(false);
  const [partToFix, setPartToFix] = useState('');
  const [lecturerNotes, setLecturerNotes] = useState('');
  const [responsibleUid, setResponsibleUid] = useState(groupMembers[0]?.uid || '');
  const [revDeadline, setRevDeadline] = useState(
    new Date(Date.now() + 3 * 86400000).toISOString().split('T')[0]
  );

  const revisions = task.revisions || [];

  const handleAddRevision = async (e) => {
    e.preventDefault();
    if (!partToFix.trim()) return;
    const respMember = groupMembers.find((m) => m.uid === responsibleUid);
    const newRev = {
      id: 'rev_' + Date.now(),
      partToFix: partToFix.trim(),
      lecturerNotes: lecturerNotes.trim(),
      responsibleUid,
      responsibleName: respMember?.name || 'Anggota Tim',
      deadline: revDeadline,
      status: 'Perlu Revisi',
      fixResultNotes: '',
      createdBy: userProfile?.name || 'Anggota',
      createdAt: new Date().toISOString()
    };

    try {
      await updateDoc(taskDocRef, {
        revisions: [...revisions, newRev],
        status: 'Perlu Revisi'
      });
      await logActivity(
        `menambahkan catatan revisi pada tugas "${task.title}" (${partToFix.trim()})`,
        'revision',
        task.id
      );
      await triggerNotification({
        title: 'Revisi Tugas Baru',
        message: `Revisi "${partToFix.trim()}" pada "${task.title}" ditugaskan ke ${
          respMember?.name || 'anggota'
        }.`,
        type: 'revision',
        targetUserIds: responsibleUid ? [responsibleUid] : [],
        taskId: task.id
      });
      setPartToFix('');
      setLecturerNotes('');
      setShowAddRev(false);
      showToast('Catatan revisi berhasil ditambahkan.');
    } catch {
      showToast('Gagal menambahkan revisi.', 'error');
    }
  };

  const handleUpdateRevision = async (revId, updates) => {
    try {
      const nextRevs = revisions.map((r) => (r.id === revId ? { ...r, ...updates } : r));
      await updateDoc(taskDocRef, { revisions: nextRevs });
      if (updates.status) {
        await logActivity(
          `memperbarui status revisi pada "${task.title}" menjadi ${updates.status}`,
          'revision',
          task.id
        );
      }
      showToast('Revisi diperbarui.');
    } catch {
      showToast('Gagal memperbarui revisi.', 'error');
    }
  };

  return (
    <div className="space-y-4">
      <div className="flex items-center justify-between">
        <div>
          <h3 className="text-base font-bold text-slate-900">Manajemen Revisi Tugas</h3>
          <p className="text-xs text-slate-500">
            Riwayat pengerjaan sebelumnya tetap tersimpan aman saat tugas dibuka kembali untuk revisi.
          </p>
        </div>
        <button
          onClick={() => setShowAddRev(!showAddRev)}
          className="px-3.5 py-2 rounded-xl bg-pink-500 hover:bg-pink-600 text-white text-xs font-semibold flex items-center gap-1.5"
        >
          <Plus className="w-4 h-4" /> Tambah Catatan Revisi
        </button>
      </div>

      {showAddRev && (
        <form
          onSubmit={handleAddRevision}
          className="bg-white rounded-2xl border border-pink-200 p-4 space-y-3 shadow-xs"
        >
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
            <div>
              <label className="block text-xs font-semibold text-slate-700 mb-1">
                Bagian yang Perlu Diperbaiki *
              </label>
              <input
                type="text"
                required
                value={partToFix}
                onChange={(e) => setPartToFix(e.target.value)}
                placeholder="Contoh: Perhitungan NPV & Tabel Proyeksi Keuangan"
                className="w-full px-3 py-2 rounded-xl border border-slate-200 text-xs"
              />
            </div>
            <div className="grid grid-cols-2 gap-2">
              <div>
                <label className="block text-xs font-semibold text-slate-700 mb-1">
                  Penanggung Jawab
                </label>
                <select
                  value={responsibleUid}
                  onChange={(e) => setResponsibleUid(e.target.value)}
                  className="w-full px-3 py-2 rounded-xl border border-slate-200 text-xs"
                >
                  {groupMembers.map((m) => (
                    <option key={m.uid} value={m.uid}>
                      {m.name}
                    </option>
                  ))}
                </select>
              </div>
              <div>
                <label className="block text-xs font-semibold text-slate-700 mb-1">
                  Deadline Revisi
                </label>
                <input
                  type="date"
                  value={revDeadline}
                  onChange={(e) => setRevDeadline(e.target.value)}
                  className="w-full px-3 py-2 rounded-xl border border-slate-200 text-xs"
                />
              </div>
            </div>
          </div>

          <div>
            <label className="block text-xs font-semibold text-slate-700 mb-1">
              Catatan Revisi dari Dosen / Tim
            </label>
            <textarea
              rows={2}
              value={lecturerNotes}
              onChange={(e) => setLecturerNotes(e.target.value)}
              placeholder="Tuliskan masukan dosen secara detail..."
              className="w-full px-3 py-2 rounded-xl border border-slate-200 text-xs"
            />
          </div>

          <div className="flex justify-end gap-2">
            <button
              type="button"
              onClick={() => setShowAddRev(false)}
              className="px-3 py-1.5 rounded-xl text-xs font-semibold text-slate-600"
            >
              Batal
            </button>
            <button
              type="submit"
              className="px-4 py-1.5 rounded-xl bg-pink-500 text-white text-xs font-semibold"
            >
              Simpan Revisi
            </button>
          </div>
        </form>
      )}

      {revisions.length === 0 ? (
        <div className="bg-white rounded-2xl border border-dashed border-slate-200 p-8 text-center">
          <p className="text-xs text-slate-500">Belum ada catatan revisi pada tugas ini.</p>
        </div>
      ) : (
        <div className="space-y-3">
          {revisions.map((rev) => (
            <div
              key={rev.id}
              className="bg-white rounded-2xl border border-slate-200 p-4 space-y-2.5"
            >
              <div className="flex flex-wrap items-start justify-between gap-2">
                <div>
                  <span
                    className={`inline-block text-[10px] font-bold px-2.5 py-0.5 rounded-full border mb-1 ${getStatusBadgeStyle(
                      rev.status
                    )}`}
                  >
                    {rev.status}
                  </span>
                  <h4 className="font-bold text-slate-900 text-sm">{rev.partToFix}</h4>
                  <p className="text-[11px] text-slate-500">
                    PJ: <strong className="text-slate-700">{rev.responsibleName}</strong> •
                    Deadline Revisi: <strong>{formatDateID(rev.deadline)}</strong>
                  </p>
                </div>

                <select
                  value={rev.status}
                  onChange={(e) => handleUpdateRevision(rev.id, { status: e.target.value })}
                  className="px-2.5 py-1 rounded-xl border border-slate-200 text-xs font-semibold"
                >
                  <option value="Perlu Revisi">Perlu Revisi</option>
                  <option value="Sedang Direvisi">Sedang Direvisi</option>
                  <option value="Selesai Direvisi">Selesai Direvisi</option>
                </select>
              </div>

              {rev.lecturerNotes && (
                <div className="bg-red-50/60 border border-red-100 p-2.5 rounded-xl text-xs text-red-900">
                  <span className="font-bold block text-[11px] text-red-700">Catatan Dosen:</span>
                  {rev.lecturerNotes}
                </div>
              )}

              <div>
                <label className="block text-[11px] font-bold text-slate-600 mb-1">
                  Catatan Hasil Perbaikan:
                </label>
                <div className="flex gap-2">
                  <input
                    type="text"
                    defaultValue={rev.fixResultNotes || ''}
                    onBlur={(e) => {
                      if (e.target.value !== (rev.fixResultNotes || '')) {
                        handleUpdateRevision(rev.id, { fixResultNotes: e.target.value });
                      }
                    }}
                    placeholder="Tuliskan hasil perbaikan yang telah dilakukan..."
                    className="flex-1 px-3 py-1.5 rounded-xl border border-slate-200 text-xs"
                  />
                </div>
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  );
}

// ============================================================================
// TASK COMMENTS & @MENTIONS SECTION
// ============================================================================
function TaskCommentsSection({
  task,
  groupMembers,
  authUser,
  userProfile,
  taskDocRef,
  logActivity,
  triggerNotification,
  showToast
}) {
  const [message, setMessage] = useState('');
  const [replyTo, setReplyTo] = useState(null);
  const comments = task.comments || [];

  const insertMention = (memberName) => {
    setMessage((prev) => `${prev}${prev && !prev.endsWith(' ') ? ' ' : ''}@${memberName} `);
  };

  const handleSendComment = async (e) => {
    e.preventDefault();
    if (!message.trim()) return;

    // Detect mentioned members
    const mentionedMembers = groupMembers.filter((m) =>
      message.toLowerCase().includes(`@${m.name.toLowerCase()}`)
    );

    const newComment = {
      id: 'cmt_' + Date.now(),
      senderId: authUser.uid,
      senderName: userProfile?.name || 'Anggota',
      avatarColor: userProfile?.avatarColor || 'bg-pink-500',
      avatarUrl: userProfile?.avatarUrl || '',
      text: message.trim(),
      replyToName: replyTo?.senderName || null,
      replyToSnippet: replyTo?.text?.slice(0, 60) || null,
      createdAt: new Date().toISOString()
    };

    try {
      await updateDoc(taskDocRef, {
        comments: [...comments, newComment]
      });

      await logActivity(`memberikan komentar pada tugas "${task.title}"`, 'comment', task.id);

      if (mentionedMembers.length > 0) {
        await triggerNotification({
          title: `${userProfile?.name} menyebut Anda (Mention)`,
          message: `Pada "${task.title}": "${message.trim()}"`,
          type: 'mention',
          targetUserIds: mentionedMembers.map((m) => m.uid),
          taskId: task.id
        });
      } else {
        await triggerNotification({
          title: 'Komentar Baru pada Tugas',
          message: `${userProfile?.name} berkomentar di "${task.title}": "${message.trim()}"`,
          type: 'comment',
          taskId: task.id
        });
      }

      setMessage('');
      setReplyTo(null);
    } catch {
      showToast('Gagal mengirim komentar.', 'error');
    }
  };

  return (
    <div className="space-y-4">
      <div>
        <h3 className="text-base font-bold text-slate-900">Diskusi & Arahan Tugas</h3>
        <p className="text-xs text-slate-500">
          Gunakan fitur @mention untuk memberi instruksi atau menanyakan progres rekan tim.
        </p>
      </div>

      {/* Quick Mention Buttons */}
      <div className="flex flex-wrap items-center gap-1.5">
        <span className="text-[11px] font-semibold text-slate-500 mr-1">Mention cepat:</span>
        {groupMembers.map((m) => (
          <button
            key={m.uid}
            type="button"
            onClick={() => insertMention(m.name)}
            className="px-2.5 py-1 rounded-full bg-pink-50 hover:bg-pink-100 text-pink-700 text-[11px] font-semibold transition"
          >
            @{m.name}
          </button>
        ))}
      </div>

      {/* Comments Feed */}
      <div className="space-y-3 max-h-80 overflow-y-auto pr-1">
        {comments.length === 0 ? (
          <div className="bg-white rounded-2xl border border-dashed border-slate-200 p-8 text-center">
            <p className="text-xs text-slate-400">
              Belum ada diskusi pada tugas ini. Mulai percakapan di bawah!
            </p>
          </div>
        ) : (
          comments.map((c) => (
            <div key={c.id} className="bg-white rounded-2xl border border-slate-100 p-3.5 shadow-2xs">
              {c.replyToName && (
                <div className="mb-2 pl-2.5 border-l-2 border-pink-300 text-[11px] text-slate-500 bg-slate-50 py-1 px-2 rounded-r-lg">
                  Membalas <strong>{c.replyToName}</strong>: "{c.replyToSnippet}..."
                </div>
              )}
              <div className="flex items-start justify-between gap-2">
                <div className="flex items-center gap-2.5">
                  <AvatarBadge
                    name={c.senderName}
                    color={c.avatarColor}
                    url={c.avatarUrl}
                    size="sm"
                  />
                  <div>
                    <span className="text-xs font-bold text-slate-900">{c.senderName}</span>
                    <span className="text-[10px] text-slate-400 ml-2">
                      {formatDateTimeID(c.createdAt)}
                    </span>
                  </div>
                </div>
                <button
                  type="button"
                  onClick={() => setReplyTo(c)}
                  className="text-[11px] font-semibold text-pink-600 hover:underline"
                >
                  Balas
                </button>
              </div>
              <p className="text-xs text-slate-700 mt-2 pl-10 whitespace-pre-line">{c.text}</p>
            </div>
          ))
        )}
      </div>

      {/* Comment Input Box */}
      <form onSubmit={handleSendComment} className="bg-white rounded-2xl border border-slate-200 p-3">
        {replyTo && (
          <div className="flex items-center justify-between bg-pink-50 px-3 py-1.5 rounded-xl mb-2 text-xs text-pink-700">
            <span>
              Membalas komentar <strong>{replyTo.senderName}</strong>
            </span>
            <button type="button" onClick={() => setReplyTo(null)}>
              <X className="w-3.5 h-3.5" />
            </button>
          </div>
        )}
        <div className="flex items-center gap-2">
          <input
            type="text"
            value={message}
            onChange={(e) => setMessage(e.target.value)}
            placeholder="Tulis instruksi, pertanyaan, atau @nama anggota..."
            className="flex-1 px-3 py-2 text-xs focus:outline-none"
          />
          <button
            type="submit"
            className="px-4 py-2 rounded-xl bg-pink-500 hover:bg-pink-600 text-white text-xs font-semibold flex items-center gap-1.5"
          >
            <Send className="w-3.5 h-3.5" /> Kirim
          </button>
        </div>
      </form>
    </div>
  );
}

// ============================================================================
// PAGE 5: KALENDER & DEADLINE
// ============================================================================
function CalendarPage({ courses, tasks, onOpenTask }) {
  const [currentDate, setCurrentDate] = useState(new Date());
  const [selectedDateStr, setSelectedDateStr] = useState(
    new Date().toISOString().split('T')[0]
  );
  const [viewMode, setViewMode] = useState('month'); // 'month' | 'week'
  const [filterCourse, setFilterCourse] = useState('ALL');
  const [filterStatus, setFilterStatus] = useState('ALL');

  const filteredTasks = useMemo(() => {
    return tasks.filter((t) => {
      if (filterCourse !== 'ALL' && t.courseId !== filterCourse) return false;
      if (filterStatus !== 'ALL' && t.status !== filterStatus) return false;
      return true;
    });
  }, [tasks, filterCourse, filterStatus]);

  // Reminders: tasks due within 3 days or overdue
  const urgentTasks = useMemo(() => {
    const now = new Date(new Date().toDateString());
    const threeDays = new Date(now.getTime() + 3 * 86400000);
    return filteredTasks.filter((t) => {
      if (t.status === 'Selesai' || !t.deadline) return false;
      const d = new Date(t.deadline);
      return d <= threeDays;
    });
  }, [filteredTasks]);

  const year = currentDate.getFullYear();
  const month = currentDate.getMonth();

  const daysInMonth = new Date(year, month + 1, 0).getDate();
  const firstDayOfWeek = new Date(year, month, 1).getDay(); // 0 = Sunday

  const calendarCells = useMemo(() => {
    const cells = [];
    for (let i = 0; i < firstDayOfWeek; i++) {
      cells.push(null);
    }
    for (let d = 1; d <= daysInMonth; d++) {
      const dateStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(d).padStart(
        2,
        '0'
      )}`;
      cells.push({ day: d, dateStr });
    }
    return cells;
  }, [year, month, daysInMonth, firstDayOfWeek]);

  // Weekly view dates around selectedDateStr
  const weekCells = useMemo(() => {
    const base = new Date(selectedDateStr || new Date());
    const dayIdx = base.getDay();
    const startOfWeek = new Date(base);
    startOfWeek.setDate(base.getDate() - dayIdx);
    const arr = [];
    for (let i = 0; i < 7; i++) {
      const d = new Date(startOfWeek);
      d.setDate(startOfWeek.getDate() + i);
      arr.push({
        day: d.getDate(),
        dayName: d.toLocaleDateString('id-ID', { weekday: 'short' }),
        dateStr: d.toISOString().split('T')[0]
      });
    }
    return arr;
  }, [selectedDateStr]);

  const tasksOnSelectedDate = filteredTasks.filter((t) => t.deadline === selectedDateStr);

  const monthTitle = currentDate.toLocaleDateString('id-ID', {
    month: 'long',
    year: 'numeric'
  });

  return (
    <div className="space-y-6">
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
        <div>
          <h1 className="text-2xl font-extrabold text-slate-900">Kalender & Deadline</h1>
          <p className="text-sm text-slate-500">
            Pantau seluruh jadwal pengumpulan tugas dari semua mata kuliah Capstone.
          </p>
        </div>

        <div className="flex items-center gap-2 bg-white p-1 rounded-2xl border border-slate-200 self-start">
          <button
            onClick={() => setViewMode('month')}
            className={`px-3.5 py-1.5 rounded-xl text-xs font-bold transition ${
              viewMode === 'month' ? 'bg-pink-500 text-white' : 'text-slate-600'
            }`}
          >
            Bulanan
          </button>
          <button
            onClick={() => setViewMode('week')}
            className={`px-3.5 py-1.5 rounded-xl text-xs font-bold transition ${
              viewMode === 'week' ? 'bg-pink-500 text-white' : 'text-slate-600'
            }`}
          >
            Mingguan
          </button>
        </div>
      </div>

      {/* Urgent Deadline Reminder Banner */}
      {urgentTasks.length > 0 && (
        <div className="bg-amber-50 border border-amber-200 rounded-2xl p-4 flex flex-wrap items-center justify-between gap-3">
          <div className="flex items-center gap-2.5">
            <AlertCircle className="w-5 h-5 text-amber-600 shrink-0" />
            <p className="text-xs text-amber-900">
              <strong>Pengingat Deadline:</strong> Terdapat{' '}
              <strong>{urgentTasks.length} tugas</strong> yang jatuh tempo dalam 3 hari ke depan
              atau sudah melewati deadline!
            </p>
          </div>
          <button
            onClick={() => onOpenTask(urgentTasks[0].id)}
            className="text-xs font-bold text-amber-800 underline"
          >
            Lihat Tugas Terdekat
          </button>
        </div>
      )}

      {/* Filters */}
      <div className="flex flex-wrap gap-3">
        <select
          value={filterCourse}
          onChange={(e) => setFilterCourse(e.target.value)}
          className="px-3.5 py-2 rounded-xl bg-white border border-slate-200 text-xs font-semibold"
        >
          <option value="ALL">Semua Mata Kuliah</option>
          {courses.map((c) => (
            <option key={c.id} value={c.id}>
              {c.name}
            </option>
          ))}
        </select>

        <select
          value={filterStatus}
          onChange={(e) => setFilterStatus(e.target.value)}
          className="px-3.5 py-2 rounded-xl bg-white border border-slate-200 text-xs font-semibold"
        >
          <option value="ALL">Semua Status</option>
          {STATUS_OPTIONS.map((s) => (
            <option key={s} value={s}>
              {s}
            </option>
          ))}
        </select>
      </div>

      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
        {/* Calendar Grid */}
        <div className="lg:col-span-2 bg-white rounded-3xl border border-slate-100 shadow-xs p-5">
          <div className="flex items-center justify-between mb-4">
            <h2 className="font-bold text-slate-900 text-base capitalize">{monthTitle}</h2>
            <div className="flex items-center gap-1.5">
              <button
                onClick={() => setCurrentDate(new Date(year, month - 1, 1))}
                className="p-2 rounded-xl hover:bg-slate-100 text-slate-600"
              >
                <ChevronLeft className="w-4 h-4" />
              </button>
              <button
                onClick={() => {
                  const now = new Date();
                  setCurrentDate(now);
                  setSelectedDateStr(now.toISOString().split('T')[0]);
                }}
                className="px-3 py-1.5 rounded-xl bg-pink-50 text-pink-700 text-xs font-bold"
              >
                Hari Ini
              </button>
              <button
                onClick={() => setCurrentDate(new Date(year, month + 1, 1))}
                className="p-2 rounded-xl hover:bg-slate-100 text-slate-600"
              >
                <ChevronRight className="w-4 h-4" />
              </button>
            </div>
          </div>

          {viewMode === 'month' ? (
            <>
              <div className="grid grid-cols-7 text-center text-[11px] font-bold text-slate-400 mb-2">
                {['Min', 'Sen', 'Sel', 'Rab', 'Kam', 'Jum', 'Sab'].map((d) => (
                  <div key={d} className="py-1">
                    {d}
                  </div>
                ))}
              </div>
              <div className="grid grid-cols-7 gap-1.5">
                {calendarCells.map((cell, i) => {
                  if (!cell) return <div key={`empty-${i}`} className="h-20 rounded-2xl" />;
                  const dayTasks = filteredTasks.filter((t) => t.deadline === cell.dateStr);
                  const isSelected = cell.dateStr === selectedDateStr;
                  const isToday = cell.dateStr === new Date().toISOString().split('T')[0];

                  return (
                    <button
                      key={cell.dateStr}
                      onClick={() => setSelectedDateStr(cell.dateStr)}
                      className={`h-20 p-1.5 rounded-2xl border text-left flex flex-col justify-between transition ${
                        isSelected
                          ? 'border-pink-500 bg-pink-50/50 ring-2 ring-pink-200'
                          : 'border-slate-100 hover:border-pink-200'
                      }`}
                    >
                      <span
                        className={`text-xs font-bold w-6 h-6 rounded-full flex items-center justify-center ${
                          isToday ? 'bg-pink-500 text-white' : 'text-slate-700'
                        }`}
                      >
                        {cell.day}
                      </span>
                      <div className="space-y-0.5 w-full overflow-hidden">
                        {dayTasks.slice(0, 2).map((dt) => (
                          <div
                            key={dt.id}
                            className={`text-[9px] font-semibold px-1.5 py-0.5 rounded truncate ${
                              dt.priority === 'Tinggi'
                                ? 'bg-red-100 text-red-700'
                                : dt.priority === 'Sedang'
                                ? 'bg-amber-100 text-amber-800'
                                : 'bg-emerald-100 text-emerald-700'
                            }`}
                          >
                            {dt.title}
                          </div>
                        ))}
                        {dayTasks.length > 2 && (
                          <span className="text-[9px] font-bold text-pink-600 block pl-1">
                            +{dayTasks.length - 2} lagi
                          </span>
                        )}
                      </div>
                    </button>
                  );
                })}
              </div>
            </>
          ) : (
            <div className="grid grid-cols-1 sm:grid-cols-7 gap-2">
              {weekCells.map((wc) => {
                const dayTasks = filteredTasks.filter((t) => t.deadline === wc.dateStr);
                const isSelected = wc.dateStr === selectedDateStr;
                return (
                  <div
                    key={wc.dateStr}
                    onClick={() => setSelectedDateStr(wc.dateStr)}
                    className={`p-3 rounded-2xl border min-h-[160px] cursor-pointer ${
                      isSelected ? 'border-pink-500 bg-pink-50/40' : 'border-slate-100'
                    }`}
                  >
                    <p className="text-[11px] font-bold text-slate-400">{wc.dayName}</p>
                    <p className="text-base font-extrabold text-slate-900 mb-2">{wc.day}</p>
                    <div className="space-y-1.5">
                      {dayTasks.map((dt) => (
                        <div
                          key={dt.id}
                          onClick={(e) => {
                            e.stopPropagation();
                            onOpenTask(dt.id);
                          }}
                          className="p-2 rounded-xl bg-white border border-slate-200 text-[11px] font-semibold text-slate-800 hover:border-pink-400"
                        >
                          {dt.title}
                        </div>
                      ))}
                    </div>
                  </div>
                );
              })}
            </div>
          )}
        </div>

        {/* Selected Date Tasks Side Panel */}
        <div className="bg-white rounded-3xl border border-slate-100 shadow-xs p-5">
          <h3 className="font-bold text-slate-900 text-base">
            Tugas pada {formatDateID(selectedDateStr)}
          </h3>
          <p className="text-xs text-slate-500 mb-4">
            Klik tanggal di kalender untuk melihat detail tugas yang jatuh tempo.
          </p>

          {tasksOnSelectedDate.length === 0 ? (
            <div className="py-10 text-center border border-dashed border-slate-200 rounded-2xl">
              <p className="text-xs text-slate-400">Tidak ada deadline tugas pada tanggal ini.</p>
            </div>
          ) : (
            <div className="space-y-3">
              {tasksOnSelectedDate.map((t) => {
                const course = courses.find((c) => c.id === t.courseId);
                return (
                  <div
                    key={t.id}
                    onClick={() => onOpenTask(t.id)}
                    className="p-3.5 rounded-2xl border border-slate-100 hover:border-pink-300 cursor-pointer transition"
                  >
                    <span className="text-[10px] font-bold text-pink-600 block">
                      {course?.name || 'Mata Kuliah'}
                    </span>
                    <h4 className="font-bold text-slate-900 text-sm mt-0.5">{t.title}</h4>
                    <div className="flex items-center justify-between mt-2">
                      <span
                        className={`text-[10px] font-bold px-2 py-0.5 rounded-full border ${getStatusBadgeStyle(
                          t.status
                        )}`}
                      >
                        {t.status}
                      </span>
                      <span className="text-xs font-bold text-pink-600">
                        {computeTaskProgress(t)}%
                      </span>
                    </div>
                  </div>
                );
              })}
            </div>
          )}
        </div>
      </div>
    </div>
  );
}

// PAGE 7: ANGGOTA KELOMPOK & PROFIL BEBAN KERJA INDIVIDU
// ============================================================================
function MembersPage({
  groupData,
  groupMembers,
  courses,
  tasks,
  onOpenMemberProfile,
  showToast
}) {
  const [copiedCode, setCopiedCode] = useState(false);
  const [copiedLink, setCopiedLink] = useState(false);

  const inviteLink = useMemo(() => {
    if (typeof window === 'undefined') return '';
    const url = new URL(window.location.href);
    url.searchParams.set('invite', groupData.inviteCode);
    return url.toString();
  }, [groupData.inviteCode]);

  const copyText = (text, type) => {
    navigator.clipboard?.writeText(text);
    if (type === 'code') {
      setCopiedCode(true);
      setTimeout(() => setCopiedCode(false), 2000);
      showToast('Kode undangan disalin!');
    } else {
      setCopiedLink(true);
      setTimeout(() => setCopiedLink(false), 2000);
      showToast('Link undangan disalin!');
    }
  };

  return (
    <div className="space-y-6">
      <div>
        <h1 className="text-2xl font-extrabold text-slate-900">Anggota Kelompok</h1>
        <p className="text-sm text-slate-500">
          Undang anggota tim dan pantau distribusi tanggung jawab serta progres masing-masing anggota.
        </p>
      </div>

      {/* Invite Banner */}
      <div className="bg-gradient-to-r from-pink-50 to-rose-50 border border-pink-200 rounded-3xl p-5 flex flex-col md:flex-row md:items-center justify-between gap-4">
        <div>
          <span className="text-xs font-bold text-pink-700 uppercase tracking-wider">
            Undang Rekan Satu Kelompok
          </span>
          <h2 className="text-lg font-extrabold text-slate-900 mt-0.5">
            Bagikan Kode atau Link Undangan
          </h2>
          <p className="text-xs text-slate-600 mt-0.5">
            Anggota baru dapat langsung bergabung dan berkolaborasi secara real-time.
          </p>
        </div>

        <div className="flex flex-wrap items-center gap-2.5">
          <button
            onClick={() => copyText(groupData.inviteCode, 'code')}
            className="px-4 py-2.5 rounded-2xl bg-white border border-pink-200 text-pink-700 font-mono font-bold text-sm flex items-center gap-2 hover:bg-pink-100/50 transition"
          >
            <span>{groupData.inviteCode}</span>
            {copiedCode ? <Check className="w-4 h-4 text-emerald-600" /> : <Copy className="w-4 h-4" />}
          </button>

          <button
            onClick={() => copyText(inviteLink, 'link')}
            className="px-4 py-2.5 rounded-2xl bg-pink-500 hover:bg-pink-600 text-white text-xs font-bold flex items-center gap-2 shadow-sm transition"
          >
            <LinkIcon className="w-4 h-4" />
            <span>{copiedLink ? 'Link Tersalin!' : 'Salin Link Undangan'}</span>
          </button>
        </div>
      </div>

      {/* Member Cards Grid */}
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-5">
        {groupMembers.map((member) => {
          const memberTasks = tasks.filter((t) =>
            (t.assignedMemberIds || []).includes(member.uid)
          );
          const completedMemberTasks = memberTasks.filter(
            (t) =>
              t.status === 'Selesai' ||
              computeMemberProgress(t.memberWork?.[member.uid]) === 100
          ).length;

          const avgMemberProgress =
            memberTasks.length === 0
              ? 0
              : Math.round(
                  memberTasks.reduce(
                    (sum, t) => sum + computeMemberProgress(t.memberWork?.[member.uid]),
                    0
                  ) / memberTasks.length
                );

          return (
            <div
              key={member.uid}
              className="bg-white rounded-3xl border border-slate-100 shadow-xs p-5 flex flex-col justify-between"
            >
              <div>
                <div className="flex items-center gap-3.5">
                  <AvatarBadge
                    name={member.name}
                    color={member.avatarColor}
                    url={member.avatarUrl}
                    size="lg"
                  />
                  <div className="min-w-0">
                    <span
                      className={`inline-block text-[10px] font-bold px-2.5 py-0.5 rounded-full mb-1 ${
                        member.role === 'Ketua Kelompok'
                          ? 'bg-pink-100 text-pink-700'
                          : 'bg-slate-100 text-slate-600'
                      }`}
                    >
                      {member.role}
                    </span>
                    <h3 className="font-bold text-slate-900 text-base truncate">{member.name}</h3>
                    <p className="text-xs text-slate-500 truncate">{member.email}</p>
                  </div>
                </div>

                <div className="mt-4">
                  <div className="flex justify-between text-xs mb-1">
                    <span className="text-slate-500">Progres Individu Keseluruhan</span>
                    <span className="font-bold text-pink-600">{avgMemberProgress}%</span>
                  </div>
                  <ProgressBar value={avgMemberProgress} />
                </div>

                <div className="grid grid-cols-2 gap-2 mt-4 pt-3 border-t border-slate-100 text-xs">
                  <div>
                    <span className="text-slate-400 block text-[11px]">Tanggung Jawab</span>
                    <span className="font-bold text-slate-800">{memberTasks.length} Tugas</span>
                  </div>
                  <div>
                    <span className="text-slate-400 block text-[11px]">Bagian Selesai</span>
                    <span className="font-bold text-emerald-600">
                      {completedMemberTasks} / {memberTasks.length}
                    </span>
                  </div>
                </div>
              </div>

              <button
                onClick={() => onOpenMemberProfile(member.uid)}
                className="mt-4 w-full py-2.5 rounded-xl bg-pink-50 hover:bg-pink-500 text-pink-700 hover:text-white text-xs font-bold transition"
              >
                Lihat Profil & Daftar Pekerjaan
              </button>
            </div>
          );
        })}
      </div>
    </div>
  );
}

// MEMBER PROFILE DETAIL PAGE (SECTION N)
// ============================================================================
function MemberProfileDetailPage({
  memberUid,
  groupMembers,
  courses,
  weeks,
  tasks,
  onBack,
  onOpenTask
}) {
  const member = groupMembers.find((m) => m.uid === memberUid);

  if (!member) {
    return (
      <div className="text-center py-12">
        <p className="text-slate-500">Profil anggota tidak ditemukan.</p>
        <button onClick={onBack} className="mt-3 text-pink-600 font-semibold text-sm">
          Kembali ke Anggota
        </button>
      </div>
    );
  }

  const memberTasks = tasks.filter((t) => (t.assignedMemberIds || []).includes(member.uid));
  const completedTasks = memberTasks.filter(
    (t) =>
      t.status === 'Selesai' || computeMemberProgress(t.memberWork?.[member.uid]) === 100
  );
  const incompleteTasks = memberTasks.filter((t) => !completedTasks.includes(t));

  const involvedCourseIds = [...new Set(memberTasks.map((t) => t.courseId))];
  const involvedCourses = courses.filter((c) => involvedCourseIds.includes(c.id));

  const overallIndividualPct =
    memberTasks.length === 0
      ? 0
      : Math.round(
          memberTasks.reduce(
            (sum, t) => sum + computeMemberProgress(t.memberWork?.[member.uid]),
            0
          ) / memberTasks.length
        );

  return (
    <div className="space-y-6">
      <button
        onClick={onBack}
        className="inline-flex items-center gap-1.5 text-xs font-bold text-slate-600 hover:text-pink-600"
      >
        <ChevronLeft className="w-4 h-4" /> Kembali ke Daftar Anggota
      </button>

      {/* Profile Header */}
      <div className="bg-white rounded-3xl border border-slate-100 shadow-xs p-6 flex flex-col sm:flex-row sm:items-center justify-between gap-4">
        <div className="flex items-center gap-4">
          <AvatarBadge
            name={member.name}
            color={member.avatarColor}
            url={member.avatarUrl}
            size="xl"
          />
          <div>
            <span className="text-xs font-bold px-3 py-0.5 rounded-full bg-pink-100 text-pink-700">
              {member.role}
            </span>
            <h1 className="text-2xl font-extrabold text-slate-900 mt-1">{member.name}</h1>
            <p className="text-xs text-slate-500">{member.email}</p>
          </div>
        </div>

        <div className="bg-pink-50/70 border border-pink-100 rounded-2xl p-4 min-w-[210px]">
          <span className="text-xs text-pink-700 font-semibold block">
            Progres Keseluruhan Individu
          </span>
          <span className="text-2xl font-extrabold text-pink-600">{overallIndividualPct}%</span>
          <ProgressBar value={overallIndividualPct} heightClass="h-2" />
        </div>
      </div>

      {/* Summary Stats */}
      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
        <div className="bg-white rounded-2xl p-4 border border-slate-100">
          <span className="text-xs text-slate-500">Mata Kuliah Terlibat</span>
          <p className="text-xl font-extrabold text-slate-900 mt-1">{involvedCourses.length}</p>
        </div>
        <div className="bg-white rounded-2xl p-4 border border-slate-100">
          <span className="text-xs text-emerald-600">Tugas Bagian Selesai</span>
          <p className="text-xl font-extrabold text-emerald-700 mt-1">{completedTasks.length}</p>
        </div>
        <div className="bg-white rounded-2xl p-4 border border-slate-100">
          <span className="text-xs text-amber-600">Tugas Belum Selesai</span>
          <p className="text-xl font-extrabold text-amber-700 mt-1">{incompleteTasks.length}</p>
        </div>
      </div>

      {/* Assigned Tasks Breakdown */}
      <div className="bg-white rounded-3xl border border-slate-100 shadow-xs p-6">
        <h2 className="text-lg font-bold text-slate-900 mb-4">
          Daftar Pekerjaan & Tanggung Jawab {member.name}
        </h2>

        {memberTasks.length === 0 ? (
          <p className="text-xs text-slate-400 text-center py-8">
            Anggota ini belum memiliki tugas yang dibagikan.
          </p>
        ) : (
          <div className="space-y-3">
            {memberTasks.map((t) => {
              const course = courses.find((c) => c.id === t.courseId);
              const week = weeks.find((w) => w.id === t.weekId);
              const mWork = t.memberWork?.[member.uid];
              const mPct = computeMemberProgress(mWork);

              return (
                <div
                  key={t.id}
                  onClick={() => onOpenTask(t.id)}
                  className="p-4 rounded-2xl border border-slate-100 hover:border-pink-300 cursor-pointer transition"
                >
                  <div className="flex flex-wrap items-center justify-between gap-2 mb-1">
                    <span className="text-xs font-bold text-pink-600">
                      {course?.name || 'Mata Kuliah'} {week ? `• ${week.title}` : ''}
                    </span>
                    <span className="text-xs font-extrabold text-slate-800">
                      Progres Individu: {mPct}%
                    </span>
                  </div>
                  <h3 className="font-bold text-slate-900 text-base">{t.title}</h3>
                  <p className="text-xs text-slate-600 mt-1">
                    <strong>Tanggung Jawab:</strong>{' '}
                    {mWork?.responsibility || 'Sesuai pembagian tugas'}
                  </p>
                  <div className="mt-2.5">
                    <ProgressBar value={mPct} heightClass="h-1.5" />
                  </div>
                </div>
              );
            })}
          </div>
        )}
      </div>
    </div>
  );
}

// ============================================================================
// PAGE 8: NOTIFICATION CENTER
// ============================================================================
function NotificationsPage({
  notifications,
  authUser,
  activeGroupId,
  onOpenTask,
  showToast
}) {
  const [filter, setFilter] = useState('ALL'); // 'ALL' | 'UNREAD' | 'READ'

  const filtered = notifications.filter((n) => {
    const isRead = (n.readBy || []).includes(authUser.uid);
    if (filter === 'UNREAD') return !isRead;
    if (filter === 'READ') return isRead;
    return true;
  });

  const markAsRead = async (notif) => {
    if ((notif.readBy || []).includes(authUser.uid)) return;
    try {
      await updateDoc(
        doc(
          db,
          'artifacts',
          appId,
          'public',
          'data',
          `grp_${activeGroupId}_notifications`,
          notif.id
        ),
        {
          readBy: arrayUnion(authUser.uid)
        }
      );
    } catch (e) {
      console.error(e);
    }
  };

  const markAllRead = async () => {
    try {
      const unread = notifications.filter((n) => !(n.readBy || []).includes(authUser.uid));
      await Promise.all(
        unread.map((n) =>
          updateDoc(
            doc(
              db,
              'artifacts',
              appId,
              'public',
              'data',
              `grp_${activeGroupId}_notifications`,
              n.id
            ),
            { readBy: arrayUnion(authUser.uid) }
          )
        )
      );
      showToast('Semua notifikasi ditandai sudah dibaca.');
    } catch {
      showToast('Gagal memperbarui notifikasi.', 'error');
    }
  };

  return (
    <div className="space-y-6">
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
        <div>
          <h1 className="text-2xl font-extrabold text-slate-900">Pusat Notifikasi</h1>
          <p className="text-sm text-slate-500">
            Pemberitahuan tugas baru, pembaruan progres, checklist, revisi, dan mention.
          </p>
        </div>

        <button
          onClick={markAllRead}
          className="px-4 py-2 rounded-xl bg-pink-50 hover:bg-pink-100 text-pink-700 text-xs font-bold self-start"
        >
          Tandai Semua Dibaca
        </button>
      </div>

      <div className="flex gap-2">
        {[
          { id: 'ALL', label: 'Semua' },
          { id: 'UNREAD', label: 'Belum Dibaca' },
          { id: 'READ', label: 'Sudah Dibaca' }
        ].map((t) => (
          <button
            key={t.id}
            onClick={() => setFilter(t.id)}
            className={`px-3.5 py-1.5 rounded-xl text-xs font-bold transition ${
              filter === t.id
                ? 'bg-pink-500 text-white'
                : 'bg-white border border-slate-200 text-slate-600'
            }`}
          >
            {t.label}
          </button>
        ))}
      </div>

      <div className="space-y-2.5">
        {filtered.length === 0 ? (
          <div className="bg-white rounded-3xl border border-dashed border-slate-200 p-10 text-center">
            <Bell className="w-8 h-8 text-slate-300 mx-auto mb-2" />
            <p className="text-xs text-slate-400">Tidak ada notifikasi pada kategori ini.</p>
          </div>
        ) : (
          filtered.map((n) => {
            const isRead = (n.readBy || []).includes(authUser.uid);
            return (
              <div
                key={n.id}
                onClick={() => {
                  markAsRead(n);
                  if (n.taskId) onOpenTask(n.taskId);
                }}
                className={`p-4 rounded-2xl border cursor-pointer transition flex items-start justify-between gap-3 ${
                  isRead
                    ? 'bg-white border-slate-100'
                    : 'bg-pink-50/60 border-pink-200 shadow-2xs'
                }`}
              >
                <div className="flex items-start gap-3">
                  <div
                    className={`w-2.5 h-2.5 rounded-full mt-1.5 shrink-0 ${
                      isRead ? 'bg-slate-200' : 'bg-pink-500'
                    }`}
                  />
                  <div>
                    <h4 className="text-xs font-bold text-slate-900">{n.title}</h4>
                    <p className="text-xs text-slate-600 mt-0.5">{n.message}</p>
                    <span className="text-[10px] text-slate-400 mt-1 block">
                      {formatDateTimeID(n.createdAt)}
                    </span>
                  </div>
                </div>
                {n.taskId && (
                  <span className="text-[11px] font-bold text-pink-600 shrink-0">Buka Tugas</span>
                )}
              </div>
            );
          })
        )}
      </div>
    </div>
  );
}

// ============================================================================
// PAGE 9: PENGATURAN (USER PROFILE & GROUP SETTINGS)
// ============================================================================
function SettingsPage({
  authUser,
  userProfile,
  groupData,
  groupMembers,
  isLeader,
  activeGroupId,
  onLeaveGroup,
  logActivity,
  showToast
}) {
  // User Profile Form
  const [name, setName] = useState(userProfile?.name || '');
  const [nim, setNim] = useState(userProfile?.nim || '');
  const [bio, setBio] = useState(userProfile?.bio || '');
  const [avatarColor, setAvatarColor] = useState(userProfile?.avatarColor || 'bg-pink-500');
  const [avatarUrl, setAvatarUrl] = useState(userProfile?.avatarUrl || '');
  const [savingProfile, setSavingProfile] = useState(false);

  // Group Settings Form
  const [groupName, setGroupName] = useState(groupData?.name || '');
  const [groupDesc, setGroupDesc] = useState(groupData?.description || '');
  const [savingGroup, setSavingGroup] = useState(false);

  const handleSaveProfile = async (e) => {
    e.preventDefault();
    if (!name.trim()) return;
    setSavingProfile(true);
    try {
      await updateDoc(doc(db, 'artifacts', appId, 'users', authUser.uid, 'profile', 'main'), {
        name: name.trim(),
        nim: nim.trim(),
        bio: bio.trim(),
        avatarColor,
        avatarUrl: avatarUrl.trim()
      });

      // Sync updated profile to group member record
      await updateDoc(
        doc(db, 'artifacts', appId, 'public', 'data', `grp_${activeGroupId}_members`, authUser.uid),
        {
          name: name.trim(),
          avatarColor,
          avatarUrl: avatarUrl.trim()
        }
      );

      showToast('Profil Anda berhasil diperbarui!');
    } catch {
      showToast('Gagal memperbarui profil.', 'error');
    } finally {
      setSavingProfile(false);
    }
  };

  const handleSaveGroup = async (e) => {
    e.preventDefault();
    if (!groupName.trim()) return;
    setSavingGroup(true);
    try {
      await updateDoc(
        doc(db, 'artifacts', appId, 'public', 'data', 'capstone_groups', activeGroupId),
        {
          name: groupName.trim(),
          description: groupDesc.trim()
        }
      );
      await logActivity(`memperbarui pengaturan kelompok "${groupName.trim()}"`, 'group');
      showToast('Pengaturan kelompok berhasil disimpan.');
    } catch {
      showToast('Gagal memperbarui pengaturan kelompok.', 'error');
    } finally {
      setSavingGroup(false);
    }
  };

  const handleRegenerateInvite = async () => {
    try {
      const newCode = generateInviteCode();
      await updateDoc(
        doc(db, 'artifacts', appId, 'public', 'data', 'capstone_groups', activeGroupId),
        { inviteCode: newCode }
      );
      showToast(`Kode undangan baru dibuat: ${newCode}`);
    } catch {
      showToast('Gagal membuat kode baru.', 'error');
    }
  };

  const handleToggleRole = async (targetMember) => {
    if (!isLeader || targetMember.uid === authUser.uid) return;
    const nextRole =
      targetMember.role === 'Ketua Kelompok' ? 'Anggota Kelompok' : 'Ketua Kelompok';
    try {
      await updateDoc(
        doc(
          db,
          'artifacts',
          appId,
          'public',
          'data',
          `grp_${activeGroupId}_members`,
          targetMember.uid
        ),
        { role: nextRole }
      );
      showToast(`Peran ${targetMember.name} diubah menjadi ${nextRole}.`);
    } catch {
      showToast('Gagal mengubah peran anggota.', 'error');
    }
  };

  const handleSwitchOrLeaveGroup = async () => {
    try {
      await updateDoc(doc(db, 'artifacts', appId, 'users', authUser.uid, 'profile', 'main'), {
        activeGroupId: null
      });
      onLeaveGroup();
    } catch {
      showToast('Gagal berpindah kelompok.', 'error');
    }
  };

  return (
    <div className="space-y-6 max-w-4xl">
      <div>
        <h1 className="text-2xl font-extrabold text-slate-900">Pengaturan Akun & Kelompok</h1>
        <p className="text-sm text-slate-500">
          Kelola identitas profil Anda serta pengaturan ruang kerja kelompok Capstone.
        </p>
      </div>

      {/* 1. Edit User Profile */}
      <div className="bg-white rounded-3xl border border-slate-100 shadow-xs p-6">
        <h2 className="text-lg font-bold text-slate-900 mb-4">Profil Pengguna</h2>
        <form onSubmit={handleSaveProfile} className="space-y-4">
          <div className="flex items-center gap-4 mb-4">
            <AvatarBadge name={name} color={avatarColor} url={avatarUrl} size="xl" />
            <div>
              <p className="text-xs font-bold text-slate-700 mb-1.5">Warna Avatar Inisial</p>
              <div className="flex gap-2">
                {AVATAR_COLORS.map((c) => (
                  <button
                    key={c}
                    type="button"
                    onClick={() => setAvatarColor(c)}
                    className={`w-6 h-6 rounded-full ${c} ${
                      avatarColor === c ? 'ring-2 ring-offset-2 ring-slate-800' : ''
                    }`}
                  />
                ))}
              </div>
            </div>
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
            <div>
              <label className="block text-xs font-semibold text-slate-700 mb-1">
                Nama Lengkap *
              </label>
              <input
                type="text"
                required
                value={name}
                onChange={(e) => setName(e.target.value)}
                className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-700 mb-1">
                NIM / ID Mahasiswa
              </label>
              <input
                type="text"
                value={nim}
                onChange={(e) => setNim(e.target.value)}
                placeholder="Contoh: 2301928374"
                className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm"
              />
            </div>
          </div>

          <div>
            <label className="block text-xs font-semibold text-slate-700 mb-1">
              URL Foto Profil (Opsional)
            </label>
            <input
              type="url"
              value={avatarUrl}
              onChange={(e) => setAvatarUrl(e.target.value)}
              placeholder="https://..."
              className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm"
            />
          </div>

          <div>
            <label className="block text-xs font-semibold text-slate-700 mb-1">Bio Singkat</label>
            <input
              type="text"
              value={bio}
              onChange={(e) => setBio(e.target.value)}
              className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm"
            />
          </div>

          <button
            type="submit"
            disabled={savingProfile}
            className="px-5 py-2.5 rounded-xl bg-pink-500 hover:bg-pink-600 text-white text-xs font-bold shadow-sm"
          >
            {savingProfile ? 'Menyimpan...' : 'Simpan Profil'}
          </button>
        </form>
      </div>

      {/* 2. Group Settings */}
      <div className="bg-white rounded-3xl border border-slate-100 shadow-xs p-6 space-y-5">
        <div className="flex items-center justify-between">
          <div>
            <h2 className="text-lg font-bold text-slate-900">Pengaturan Kelompok</h2>
            <p className="text-xs text-slate-500">
              Kode Undangan Aktif:{' '}
              <strong className="font-mono text-pink-600">{groupData.inviteCode}</strong>
            </p>
          </div>
          {isLeader && (
            <button
              type="button"
              onClick={handleRegenerateInvite}
              className="px-3 py-1.5 rounded-xl bg-slate-100 hover:bg-slate-200 text-slate-700 text-xs font-bold"
            >
              Reset Kode Undangan
            </button>
          )}
        </div>

        <form onSubmit={handleSaveGroup} className="space-y-4">
          <div>
            <label className="block text-xs font-semibold text-slate-700 mb-1">
              Nama Kelompok
            </label>
            <input
              type="text"
              disabled={!isLeader}
              value={groupName}
              onChange={(e) => setGroupName(e.target.value)}
              className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm disabled:bg-slate-50"
            />
          </div>

          <div>
            <label className="block text-xs font-semibold text-slate-700 mb-1">
              Deskripsi Kelompok
            </label>
            <textarea
              rows={2}
              disabled={!isLeader}
              value={groupDesc}
              onChange={(e) => setGroupDesc(e.target.value)}
              className="w-full px-3.5 py-2.5 rounded-xl border border-slate-200 text-sm disabled:bg-slate-50"
            />
          </div>

          {isLeader && (
            <button
              type="submit"
              disabled={savingGroup}
              className="px-5 py-2.5 rounded-xl bg-slate-900 hover:bg-pink-600 text-white text-xs font-bold transition"
            >
              {savingGroup ? 'Menyimpan...' : 'Simpan Pengaturan Kelompok'}
            </button>
          )}
        </form>

        {/* Member Role Management (Leader Only) */}
        {isLeader && (
          <div className="pt-4 border-t border-slate-100">
            <h3 className="text-sm font-bold text-slate-800 mb-3">Kelola Peran Anggota</h3>
            <div className="space-y-2">
              {groupMembers.map((m) => (
                <div
                  key={m.uid}
                  className="flex items-center justify-between p-3 rounded-2xl bg-slate-50"
                >
                  <div className="flex items-center gap-2.5">
                    <AvatarBadge name={m.name} color={m.avatarColor} url={m.avatarUrl} size="sm" />
                    <div>
                      <p className="text-xs font-bold text-slate-900">{m.name}</p>
                      <p className="text-[11px] text-slate-500">{m.role}</p>
                    </div>
                  </div>
                  {m.uid !== authUser.uid && (
                    <button
                      type="button"
                      onClick={() => handleToggleRole(m)}
                      className="px-3 py-1 rounded-xl bg-white border border-slate-200 text-xs font-semibold text-slate-700 hover:border-pink-400"
                    >
                      Ubah ke {m.role === 'Ketua Kelompok' ? 'Anggota' : 'Ketua'}
                    </button>
                  )}
                </div>
              ))}
            </div>
          </div>
        )}

        <div className="pt-4 border-t border-slate-100 flex items-center justify-between">
          <div>
            <p className="text-xs font-bold text-slate-800">Ganti / Buat Kelompok Lain</p>
            <p className="text-[11px] text-slate-500">
              Kembali ke halaman pemilihan kelompok untuk membuat atau bergabung kelompok baru.
            </p>
          </div>
          <button
            type="button"
            onClick={handleSwitchOrLeaveGroup}
            className="px-4 py-2 rounded-xl border border-red-200 text-red-600 hover:bg-red-50 text-xs font-bold"
          >
            Ganti Kelompok
          </button>
        </div>
      </div>
    </div>
  );
}

function ConfirmDialog({ title, message, confirmText = 'Konfirmasi', onConfirm, onCancel }) {
  return (
    <div className="fixed inset-0 z-[70] bg-slate-900/50 backdrop-blur-xs flex items-center justify-center p-4">
      <div className="bg-white rounded-3xl max-w-sm w-full p-6 shadow-2xl border border-slate-100">
        <h3 className="text-lg font-bold text-slate-900">{title}</h3>
        <p className="text-xs text-slate-600 mt-2 leading-relaxed">{message}</p>
        <div className="flex justify-end gap-2 mt-6">
          <button
            onClick={onCancel}
            className="px-4 py-2 rounded-xl text-xs font-semibold text-slate-600 hover:bg-slate-100"
          >
            Batal
          </button>
          <button
            onClick={onConfirm}
            className="px-4 py-2 rounded-xl bg-red-600 hover:bg-red-700 text-white text-xs font-semibold"
          >
            {confirmText}
          </button>
        </div>
      </div>
    </div>
  );
}
